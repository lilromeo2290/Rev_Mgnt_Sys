    71→// ─── Component ───────────────────────────────────────────────────────────────
    72→
    73→export function BusinessesPage() {
    74→  const [view, setView] = useState<'list' | 'form'>('list');
    75→  const [searchQuery, setSearchQuery] = useState('');
    76→  const [statusFilter, setStatusFilter] = useState<string>('All');
    77→  const [typeFilter, setTypeFilter] = useState<string>('All');
    78→  const [currentPage, setCurrentPage] = useState(1);
    79→  const [businesses, setBusinesses] = useState<Business[]>(mockBusinesses);
    80→  const itemsPerPage = 10;
    81→
    82→  // ── Form State ───────────────────────────────────────────────────────────
    83→  const [form, setForm] = useState({
    84→    regNumber: '',
    85→    name: '',
    86→    type: '',
    87→    category: '',
    88→    subCategory: '',
    89→    tin: '',
    90→    licenseNumber: '',
    91→    dateRegistered: '',
    92→    status: 'Active',
    93→    ownerName: '',
    94→    ghanaCard: '',
    95→    phone: '',
    96→    email: '',
    97→    gpsAddress: '',
    98→    digitalAddress: '',
    99→    residentialAddress: '',
   100→    businessAddress: '',
   101→    ward: '',
   102→    electoralArea: '',
   103→    zone: '',
   104→    revenueArea: '',
   105→  });

   125→  const safeCurrentPage = Math.min(currentPage, totalPages);
   126→  const startIdx = (safeCurrentPage - 1) * itemsPerPage;
   127→  const paged = filtered.slice(startIdx, startIdx + itemsPerPage);
   128→  const showingFrom = filtered.length === 0 ? 0 : startIdx + 1;
   129→  const showingTo = Math.min(startIdx + itemsPerPage, filtered.length);

   132→  const handleFormChange = (
   133→    e: React.ChangeEvent<
   134→      HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement

   137→    const { name, value } = e.target;
   138→    setForm((prev) => {
   139→      const updated = { ...prev, [name]: value };
   140→      // Reset category and sub-category when type changes
   141→      if (name === 'type') {
   142→        updated.category = '';
   143→        updated.subCategory = '';

   145→      return updated;
   146→    });
   147→  };

   149→  const handleSave = () => {
   150→    setView('list');
   151→  };
   152→
   153→  const handleCancel = () => {
   154→    setView('list');
   155→  };
   156→
   157→  const handleDelete = (regNumber: string) => {
   158→    setBusinesses((prev) => prev.filter((b) => b.regNumber !== regNumber));

   165→    'text-sm font-medium text-slate-700 dark:text-slate-300 mb-1.5';

   173→    return (

   178→            <h1 className="text-2xl font-bold text-slate-900 dark:text-white">
   179→              Business Registration
   180→            </h1>
   181→            <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">

   184→            </p>

   186→          <button
   187→            onClick={() => setView('form')}

   190→            <Plus className="w-4 h-4" />
   191→            Register New Business
   192→          </button>
   193→        </div>
   194→
   195→        {/* ── Search & Filters ────────────────────────────────────────────── */}
   196→        <div className="flex flex-col sm:flex-row gap-3">
   197→          <div className="relative flex-1">
   198→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />

   201→              placeholder="Search by name, owner, or registration number..."

   239→        {/* ── Table ───────────────────────────────────────────────────────── */}

   251→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">

   292→                        {biz.name}
   293→                      </td>
   294→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap">
   295→                        {biz.owner}
   296→                      </td>

   298→                        {biz.type}
   299→                      </td>

   301→                        {biz.category}
   302→                      </td>

   306→                      <td className="px-4 py-3 whitespace-nowrap">
   307→                        <span
   308→                          className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-semibold ${
   309→                            biz.status === 'Active'
   310→                              ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400'
   311→                              : 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400'
   312→                          }`}
   313→                        >
   314→                          {biz.status}
   315→                        </span>
   316→                      </td>
   317→                      <td className="px-4 py-3 text-right whitespace-nowrap">
   318→                        <div className="inline-flex items-center gap-1">
   319→                          <button
   320→                            className="p-1.5 rounded-md text-slate-400 hover:text-emerald-600 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   321→                            title="Edit"
   322→                          >
   323→                            <Pencil className="w-4 h-4" />
   324→                          </button>
   325→                          <button
   326→                            onClick={() => handleDelete(biz.regNumber)}
   327→                            className="p-1.5 rounded-md text-slate-400 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-900/20 transition-colors"
   328→                            title="Delete"
   329→                          >
   330→                            <Trash2 className="w-4 h-4" />
   331→                          </button>
   332→                        </div>
   333→                      </td>
   334→                    </tr>
   335→                  ))
   336→                )}
   337→              </tbody>
   338→            </table>
   339→          </div>
   340→        </div>
   341→
   342→        {/* ── Pagination ──────────────────────────────────────────────────── */}
   343→        <div className="flex flex-col sm:flex-row items-center justify-between gap-3 text-sm">
   344→          <p className="text-slate-500 dark:text-slate-400">
   345→            Showing {showingFrom}-{showingTo} of {filtered.length}
   346→          </p>
   347→          <div className="flex items-center gap-2">
   348→            <button
   349→              onClick={() => setCurrentPage((p) => Math.max(1, p - 1))}
   350→              disabled={safeCurrentPage <= 1}
   351→              className="inline-flex items-center gap-1 px-3 py-1.5 rounded-lg border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-800 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   352→            >
   353→              <ChevronLeft className="w-4 h-4" />
   354→              Previous
   355→            </button>
   356→            <span className="px-3 py-1.5 rounded-lg bg-slate-100 dark:bg-slate-800 text-slate-700 dark:text-slate-300 font-medium">
   357→              {safeCurrentPage} / {totalPages}
   358→            </span>
   359→            <button
   360→              onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))}
   361→              disabled={safeCurrentPage >= totalPages}
   362→              className="inline-flex items-center gap-1 px-3 py-1.5 rounded-lg border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-800 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   363→            >
   364→              Next
   365→              <ChevronRight className="w-4 h-4" />
   366→            </button>
   367→          </div>
   368→        </div>
   369→      </div>
   370→    );
   371→  }
   372→

   374→  //  FORM VIEW
   375→  // ══════════════════════════════════════════════════════════════════════════
   376→  return (
   377→    <div className="space-y-6">
   378→      {/* ── Header ────────────────────────────────────────────────────────── */}
   379→      <div className="flex items-center gap-4">
   380→        <button
   381→          onClick={handleCancel}
   382→          className="p-2 rounded-lg border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors"
   383→        >
   384→          <ArrowLeft className="w-5 h-5" />
   385→        </button>
   386→        <div>
   387→          <h1 className="text-2xl font-bold text-slate-900 dark:text-white">
   388→            Register New Business
   389→          </h1>
   390→          <p className="text-sm text-slate-500 dark:text-slate-400 mt-0.5">
   391→            Fill in the details below to register a new business.
   392→          </p>
   393→        </div>
   394→      </div>
   395→
   396→      <div className="space-y-8">
   397→        {/* ─── Section 1: Business Details ──────────────────────────────── */}
   398→        <section>
   399→          <h2 className={sectionHeaderClass}>Business Details</h2>
   400→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
   401→            {/* Business Registration Number */}
   402→            <div>
   403→              <label className={`${labelClass} block`}>
   404→                Business Registration Number
   405→              </label>
   406→              <input
   407→                type="text"
   408→                name="regNumber"
   409→                value={form.regNumber}
   410→                onChange={handleFormChange}
   411→                placeholder="e.g. BIZ-2024-013"
   412→                className={inputClass}
   413→              />
   414→            </div>
   415→
   416→            {/* Business Name */}
   417→            <div>
   418→              <label className={`${labelClass} block`}>Business Name</label>
   419→              <input
   420→                type="text"
   421→                name="name"
   422→                value={form.name}
   423→                onChange={handleFormChange}
   424→                placeholder="Enter business name"
   425→                className={inputClass}
   426→              />
   427→            </div>
   428→
   429→            {/* Business Type */}
   430→            <div>
   431→              <label className={`${labelClass} block`}>Business Type</label>
   432→              <select
   433→                name="type"
   434→                value={form.type}
   435→                onChange={handleFormChange}
   436→                className={inputClass}
   437→              >
   438→                <option value="">Select business type</option>
   439→                {businessTypes.map((t) => (
   440→                  <option key={t} value={t}>

   446→
