   186→  const [statusFilter, setStatusFilter] = useState<'All' | ReceiptStatus>('All');

   189→  const [selectedReceipt, setSelectedReceipt] = useState<Receipt | null>(null);
   190→  const [showFilters, setShowFilters] = useState(false);
   191→  const [copiedId, setCopiedId] = useState<string | null>(null);
   192→
   193→  const filtered = useMemo(() => {
   194→    return mockReceipts.filter((r) => {
   195→      const q = search.toLowerCase();
   196→      const matchSearch =
   197→        !q ||
   198→        r.receiptNo.toLowerCase().includes(q) ||
   199→        r.issuedTo.toLowerCase().includes(q) ||
   200→        r.issuedBy.toLowerCase().includes(q) ||
   201→        r.billNo.toLowerCase().includes(q);
   202→      const matchStatus = statusFilter === 'All' || r.status === statusFilter;
   203→      const matchMethod = methodFilter === 'All' || r.method === methodFilter;
   204→      return matchSearch && matchStatus && matchMethod;
   205→    });
   206→  }, [search, statusFilter, methodFilter]);
   207→
   208→  const totalPages = Math.max(1, Math.ceil(filtered.length / ITEMS_PER_PAGE));
   209→  const paginated = filtered.slice((page - 1) * ITEMS_PER_PAGE, page * ITEMS_PER_PAGE);

   212→    navigator.clipboard.writeText(text).catch(() => {});

   246→            <p className="text-sm text-slate-500 dark:text-slate-400">Total Receipts</p>

   376→                    </td>

   378→                      <span className={`inline-flex text-xs font-medium px-2 py-0.5 rounded-full ${statusStyle[r.status].pill}`}>
   379→                        {r.status}
   380→                      </span>
   381→                    </td>
   382→                    <td className="px-4 py-3">
   383→                      <div className="flex items-center justify-center gap-1">
   384→                        <button
   385→                          onClick={() => setSelectedReceipt(r)}
   386→                          className="p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   387→                          title="View details"
   388→                        >
   389→                          <Eye className="w-4 h-4" />
   390→                        </button>
   391→                        <button
   392→                          className="p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   393→                          title="Print receipt"
   394→                        >
   395→                          <Printer className="w-4 h-4" />
   396→                        </button>
   397→                      </div>
   398→                    </td>
   399→                  </tr>
   400→                ))
   401→              )}
   402→            </tbody>
   403→          </table>
   404→        </div>
   405→
   406→        {/* Pagination */}
   407→        <div className="flex items-center justify-between px-4 py-3 border-t border-slate-200 dark:border-slate-700">
   408→          <p className="text-sm text-slate-500 dark:text-slate-400">
   409→            Showing {paginated.length} of {filtered.length} receipts
   410→          </p>
   411→          <div className="flex items-center gap-1">
   412→            <button
   413→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   414→              disabled={page === 1}
   415→              className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   416→            >
   417→              <ChevronLeft className="w-4 h-4" />
   418→            </button>
   419→            {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
   420→              <button
   421→                key={p}
   422→                onClick={() => setPage(p)}
   423→                className={`w-8 h-8 rounded-lg text-sm font-medium transition-colors ${
   424→                  p === page
   425→                    ? 'bg-emerald-600 text-white'
   426→                    : 'text-slate-500 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-700'
   427→                }`}
   428→              >
   429→                {p}
   430→              </button>
   431→            ))}
   432→            <button
   433→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   434→              disabled={page === totalPages}
   435→              className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   436→            >
   437→              <ChevronRight className="w-4 h-4" />
   438→            </button>
   439→          </div>
   440→        </div>
   441→      </div>
   442→
   443→      {/* Detail Modal */}
   444→      {selectedReceipt && (
   445→        <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
   446→          <div className="fixed inset-0 bg-black/50 backdrop-blur-sm" onClick={() => setSelectedReceipt(null)} />
   447→          <div className="relative bg-white dark:bg-slate-800 rounded-xl border border-slate-200 dark:border-slate-700 shadow-2xl w-full max-w-lg max-h-[90vh] overflow-y-auto">
   448→            {/* Modal Header */}
   449→            <div className="flex items-center justify-between p-5 border-b border-slate-200 dark:border-slate-700">
   450→              <div className="flex items-center gap-3">
   451→                <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   452→                  <Receipt className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   453→                </div>
   454→                <div>
   455→                  <h2 className="text-lg font-semibold text-slate-900 dark:text-white">Receipt Details</h2>
   456→                  <p className="text-sm font-mono text-slate-500 dark:text-slate-400">{selectedReceipt.receiptNo}</p>
   457→                </div>
   458→              </div>
   459→              <button
   460→                onClick={() => setSelectedReceipt(null)}
   461→                className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 transition-colors"
   462→              >
   463→                <X className="w-5 h-5" />
   464→              </button>
   465→            </div>
   466→
   467→            {/* Modal Body */}
   468→            <div className="p-5 space-y-5">
   469→              {/* Status Badge */}
   470→              <div className="flex items-center justify-between">
   471→                <span className={`inline-flex items-center gap-1.5 text-xs font-semibold px-2.5 py-1 rounded-full ${statusStyle[selectedReceipt.status].pill}`}>
   472→                  {selectedReceipt.status}
   473→                </span>
   474→                <div className="flex items-center gap-2">
   475→                  <button className="inline-flex items-center gap-1.5 rounded-lg bg-emerald-600 text-white px-3 py-1.5 text-sm font-medium hover:bg-emerald-700 transition-colors">
   476→                    <Printer className="w-3.5 h-3.5" />
   477→                    Print
   478→                  </button>
   479→                  <button className="inline-flex items-center gap-1.5 rounded-lg border border-slate-200 dark:border-slate-700 text-slate-700 dark:text-slate-300 px-3 py-1.5 text-sm font-medium hover:bg-slate-50 dark:hover:bg-slate-700 transition-colors">
   480→                    <Download className="w-3.5 h-3.5" />
   481→                    PDF
   482→                  </button>
   483→                </div>
   484→              </div>
   485→
   486→              {/* Info Grid */}
   487→              <div className="grid grid-cols-2 gap-4">
   488→                <InfoField label="Bill Reference" value={selectedReceipt.billNo} />
   489→                <InfoField label="Payment Method" value={selectedReceipt.method} />
   490→                <InfoField label="Transaction Ref" value={selectedReceipt.reference} />
   491→                <InfoField label="Zone / Ward" value={`${selectedReceipt.zone} — ${selectedReceipt.ward}`} />
   492→                <InfoField label="Collected By" value={selectedReceipt.issuedBy} />
   493→                <InfoField label="Entity Type" value={selectedReceipt.entityType} />
   494→              </div>
   495→
   496→              {/* Issued To */}
   497→              <div className="rounded-lg bg-slate-50 dark:bg-slate-900/50 p-4">
   498→                <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium mb-1">Issued To</p>
   499→                <p className="text-lg font-semibold text-slate-900 dark:text-white">{selectedReceipt.issuedTo}</p>
   500→              </div>
   501→
   502→              {/* Line Items */}
   503→              <div>
   504→                <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium mb-3">Line Items</p>
   505→                <div className="rounded-lg border border-slate-200 dark:border-slate-700 overflow-hidden">
   506→                  <table className="w-full text-left text-sm">
   507→                    <thead className="bg-slate-50 dark:bg-slate-900/50">
   508→                      <tr className="border-b border-slate-200 dark:border-slate-700">
   509→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400">Description</th>
   510→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Qty</th>
   511→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Unit Price</th>
   512→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Amount</th>
   513→                      </tr>
   514→                    </thead>
   515→                    <tbody className="divide-y divide-slate-100 dark:divide-slate-700/60">
   516→                      {selectedReceipt.items.map((item, idx) => (
   517→                        <tr key={idx}>
   518→                          <td className="px-3 py-2 text-slate-700 dark:text-slate-300">{item.description}</td>
   519→                          <td className="px-3 py-2 text-slate-600 dark:text-slate-400 text-right">{item.quantity}</td>
   520→                          <td className="px-3 py-2 text-slate-600 dark:text-slate-400 text-right">{fmtCurrency(item.unitPrice)}</td>
   521→                          <td className="px-3 py-2 text-slate-900 dark:text-white font-medium text-right">{fmtCurrency(item.amount)}</td>
   522→                        </tr>
   523→                      ))}
   524→                    </tbody>
   525→                    <tfoot className="bg-slate-50 dark:bg-slate-900/50">
   526→                      <tr className="border-t border-slate-200 dark:border-slate-700">
   527→                        <td colSpan={3} className="px-3 py-2.5 text-sm font-semibold text-slate-700 dark:text-slate-300 text-right">Total Paid</td>
   528→                        <td className="px-3 py-2.5 text-sm font-bold text-emerald-600 dark:text-emerald-400 text-right">{fmtCurrency(selectedReceipt.totalPaid)}</td>
   529→                      </tr>
   530→                    </tfoot>
   531→                  </table>
   532→                </div>
   533→              </div>
   534→
   535→              {/* Date / Time */}
   536→              <div className="flex items-center gap-6 text-sm text-slate-500 dark:text-slate-400">
   537→                <span className="flex items-center gap-1.5"><CalendarCheck className="w-4 h-4" />{selectedReceipt.date}</span>
   538→                <span className="flex items-center gap-1.5"><UserCheck className="w-4 h-4" />{selectedReceipt.issuedBy}</span>
   539→              </div>
   540→            </div>
   541→          </div>
   542→        </div>
   543→      )}
   544→    </div>
   545→  );
   546→}
