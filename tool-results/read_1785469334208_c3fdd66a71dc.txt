   104→function readLS<T>(key: string, fallback: T): T {
   105→  if (typeof window === 'undefined') return fallback;
   106→  try {
   107→    const raw = window.localStorage.getItem(key);
   108→    if (raw) return JSON.parse(raw) as T;
   109→  } catch { /* ignore */ }
   110→  return fallback;
   111→}

   219→/* ---------- Custom Recharts Tooltip ---------- */
   220→
   221→function RevenueTooltip({ active, payload, label }: any) {
   222→  if (!active || !payload || !payload.length) return null;
   223→  return (
   224→    <div className="rounded-lg border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-800 shadow-lg px-3 py-2 text-sm">
   225→      <p className="font-medium text-slate-900 dark:text-white">{label}</p>
   226→      {payload.map((entry: any, i: number) => (
   227→        <p key={i} style={{ color: entry.color }} className="font-semibold">
   228→          {entry.name}: {fmtCurrency(entry.value)}
   229→        </p>
   230→      ))}
   231→    </div>
   232→  );
   233→}

   235→/* ---------- Pie Custom Label ---------- */
   236→
   237→const RADIAN = Math.PI / 180;
   238→function renderPieLabel({ cx, cy, midAngle, innerRadius, outerRadius, percent, name }: any) {
   239→  const radius = innerRadius + (outerRadius - innerRadius) * 1.4;
   240→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   241→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   242→
   243→  if (percent < 0.05) return null;
   244→
   245→  return (
   246→    <text x={x} y={y} fill="#475569" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[11px]">
   247→      {name || `${(percent * 100).toFixed(0)}%`}
   248→    </text>
   249→  );
   250→}

   267→export function DashboardPage() {
   268→  // Read all data from localStorage on mount + listen for changes
   269→  const [businesses, setBusinesses] = useState<LSBusiness[]>([]);
   270→  const [properties, setProperties] = useState<LSProperty[]>([]);
   271→  const [bills, setBills] = useState<LSBill[]>([]);
   272→  const [payments, setPayments] = useState<LSPayment[]>([]);
   273→
   274→  useEffect(() => {
   275→    // Initial load
   276→    setBusinesses(readLS<LSBusiness[]>('rms-businesses', []));
   277→    setProperties(readLS<LSProperty[]>('rms-properties', []));
   278→    setBills(readLS<LSBill[]>('rms-bills', []));
   279→    setPayments(readLS<LSPayment[]>('rms-payments', []));
   280→
   281→    // Listen for storage changes (cross-tab sync)
   282→    const handler = () => {
   283→      setBusinesses(readLS<LSBusiness[]>('rms-businesses', []));
   284→      setProperties(readLS<LSProperty[]>('rms-properties', []));
   285→      setBills(readLS<LSBill[]>('rms-bills', []));
   286→      setPayments(readLS<LSPayment[]>('rms-payments', []));
   287→    };
   288→    window.addEventListener('storage', handler);
   289→    // Also poll every 3 seconds so same-tab updates reflect
   290→    const interval = setInterval(handler, 3000);
   291→    return () => {
   292→      window.removeEventListener('storage', handler);
   293→      clearInterval(interval);
   294→    };
   295→  }, []);
   296→
   297→  // ── Computed Stats ────────────────────────────────────────────────────
   298→
   299→  const totalBusinesses = businesses.length;
   300→  const activeBusinesses = businesses.filter(b => b.status === 'Active').length;
   301→  const totalProperties = properties.length;
   302→  
   303→  const totalCollected = payments.reduce((sum, p) => sum + (p.amount || 0), 0);
   304→  const totalBilled = bills.reduce((sum, b) => sum + (b.totalDue || 0), 0);
   305→  const totalPaidBills = bills.filter(b => b.status === 'Paid').reduce((sum, b) => sum + (b.totalDue || 0), 0);
   306→  const outstanding = bills.reduce((sum, b) => {
   307→    if (b.status === 'Paid') return sum;
   308→    return sum + (b.totalDue || 0);
   309→  }, 0);
   310→
   311→  // Revenue by month (from payments)
   312→  const monthlyRevenue = useMemo(() => {
   313→    const map = new Map<string, number>();
   314→    payments.forEach(p => {
   315→      if (!p.date) return;
   316→      // Extract YYYY-MM from date string
   317→      const parts = p.date.split('-');
   318→      if (parts.length >= 2) {
   319→        const key = `${parts[0]}-${parts[1]}`;
   320→        map.set(key, (map.get(key) || 0) + (p.amount || 0));
   321→      }
   322→    });
   323→    // Sort by month
   324→    const entries = Array.from(map.entries()).sort((a, b) => a[0].localeCompare(b[0]));
   325→    return entries.map(([month, revenue]) => {
   326→      // Format month label
   327→      const [y, m] = month.split('-');
   328→      const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
   329→      return {
   330→        month: `${monthNames[parseInt(m, 10) - 1]} ${y}`,
   331→        revenue,
   332→      };
   333→    });
   334→  }, [payments]);
   335→
   336→  // Revenue by category (from bills by revenue item)
   337→  const revenueByCategory = useMemo(() => {
   338→    const map = new Map<string, number>();
   339→    bills.forEach(b => {
   340→      const cat = b.revenueItem || b.category || 'Other';
   341→      map.set(cat, (map.get(cat) || 0) + (b.totalDue || 0));
   342→    });
   343→    return Array.from(map.entries())
   344→      .map(([name, value], i) => ({
   345→        name,
   346→        value,
   347→        color: COLORS[i % COLORS.length],
   348→      }))
   349→      .sort((a, b) => b.value - a.value);
   350→  }, [bills]);
   351→
   352→  // Businesses by category (pie chart)
   353→  const businessesByCategory = useMemo(() => {
   354→    const map = new Map<string, number>();
   355→    businesses.forEach(b => {
   356→      const cat = b.category || b.type || 'Uncategorized';
   357→      map.set(cat, (map.get(cat) || 0) + 1);
   358→    });
   359→    return Array.from(map.entries())
   360→      .map(([name, value], i) => ({
   361→        name,
   362→        value,
   363→        color: COLORS[i % COLORS.length],
   364→      }))
   365→      .sort((a, b) => b.value - a.value);
   366→  }, [businesses]);
   367→
   368→  // Businesses by status
   369→  const businessesByStatus = useMemo(() => {
   370→    const active = businesses.filter(b => b.status === 'Active').length;
   371→    const inactive = businesses.filter(b => b.status === 'Inactive').length;
   372→    return [
   373→      { name: 'Active', value: active, color: '#10b981' },
   374→      { name: 'Inactive', value: inactive, color: '#ef4444' },
   375→    ];
   376→  }, [businesses]);
   377→
   378→  // Bills by status
   379→  const billsByStatus = useMemo(() => {
   380→    const statusMap = new Map<string, number>();
   381→    bills.forEach(b => {
   382→      const s = b.status || 'Unknown';
   383→      statusMap.set(s, (statusMap.get(s) || 0) + 1);
   384→    });
   385→    const statusColors: Record<string, string> = {
   386→      Paid: '#10b981',
   387→      Unpaid: '#ef4444',
   388→      Partial: '#f59e0b',
   389→      Overdue: '#dc2626',
   390→    };
   391→    return Array.from(statusMap.entries()).map(([name, value]) => ({
   392→      name,
   393→      value,
   394→      color: statusColors[name] || '#64748b',
   395→    }));
   396→  }, [bills]);
   397→
   398→  // Recent payments (last 10)
   399→  const recentPayments = useMemo(() => {
   400→    return [...payments]
   401→      .sort((a, b) => (b.date || '').localeCompare(a.date || ''))
   402→      .slice(0, 10);
   403→  }, [payments]);
   404→
   405→  // Top collectors (from payments grouped by collector)
   406→  const topCollectors = useMemo(() => {
   407→    const map = new Map<string, { amount: number; bills: number; area: string }>();
   408→    payments.forEach(p => {
   409→      const name = p.collector || 'Unknown';
   410→      const existing = map.get(name) || { amount: 0, bills: 0, area: '' };
   411→      existing.amount += p.amount || 0;
   412→      existing.bills += 1;
   413→      existing.area = p.method || '';
   414→      map.set(name, existing);
   415→    });
   416→    return Array.from(map.entries())
   417→      .map(([name, data]) => ({ name, ...data }))
   418→      .sort((a, b) => b.amount - a.amount)
   419→      .slice(0, 8);
   420→  }, [payments]);
   421→
   422→  // Recent businesses registered (last 8)
   423→  const recentBusinesses = useMemo(() => {
   424→    return [...businesses]
   425→      .sort((a, b) => (b.dateRegistered || '').localeCompare(a.dateRegistered || ''))
   426→      .slice(0, 8);
   427→  }, [businesses]);
   428→
   429→  // Collection rate
   430→  const collectionRate = totalBilled > 0 ? Math.round((totalPaidBills / totalBilled) * 100) : 0;
   431→
   432→  // Determine if we have any data at all
   433→  const hasData = totalBusinesses > 0 || totalProperties > 0 || bills.length > 0 || payments.length > 0;
   434→
   435→  return (
   436→    <div className="min-h-screen bg-slate-50 dark:bg-slate-900">
   437→      {/* Header */}
   438→      <header className="bg-white dark:bg-slate-800 border-b border-slate-200 dark:border-slate-700 px-6 py-4">
   439→        <div className="max-w-7xl mx-auto flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   440→          <div className="flex items-center gap-3">
   441→            <div className="w-9 h-9 rounded-lg bg-emerald-600 flex items-center justify-center">
   442→              <Landmark className="w-5 h-5 text-white" />
   443→            </div>
   444→            <div>
   445→              <h1 className="text-xl font-bold text-slate-900 dark:text-white">
   446→                Revenue Management System
   447→              </h1>
   448→              <p className="text-xs text-slate-500 dark:text-slate-400">
   449→                Kumasi Metropolitan Assembly — Dashboard Overview
   450→              </p>
   451→            </div>
   452→          </div>
   453→          <div className="flex items-center gap-4">
   454→            <div className="text-right hidden sm:block">
   455→              <p className="text-xs text-slate-400 dark:text-slate-500">Collection Rate</p>
   456→              <p className={`text-lg font-bold ${collectionRate >= 50 ? 'text-emerald-600' : collectionRate > 0 ? 'text-amber-600' : 'text-slate-400'}`}>
   457→                {collectionRate}%
   458→              </p>
   459→            </div>
   460→            <p className="text-sm text-slate-500 dark:text-slate-400">
   461→              Fiscal Year 2024
   462→            </p>
   463→          </div>
   464→        </div>
   465→      </header>
   466→
   467→      <main className="max-w-7xl mx-auto px-4 sm:px-6 py-6 space-y-6">
   468→        {/* ── Stat Cards ──────────────────────────────────────────────── */}
   469→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   470→          <StatCard
   471→            icon={<Building2 className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />}
   472→            label="Total Businesses"
   473→            value={fmt(totalBusinesses)}
   474→            change={totalBusinesses > 0 ? 12 : 0}
   475→            accent="bg-emerald-100 dark:bg-emerald-900/40"
   476→            sub={`${activeBusinesses} active`}
   477→          />
   478→          <StatCard
   479→            icon={<Home className="w-5 h-5 text-blue-600 dark:text-blue-400" />}
   480→            label="Total Properties"
   481→            value={fmt(totalProperties)}
   482→            accent="bg-blue-100 dark:bg-blue-900/40"
   483→          />
   484→          <StatCard
   485→            icon={<CircleDollarSign className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />}
   486→            label="Amount Collected"
   487→            value={fmtCurrency(totalCollected)}
   488→            change={totalCollected > 0 ? 8 : 0}
   489→            accent="bg-emerald-100 dark:bg-emerald-900/40"
   490→            sub={`${payments.length} payment(s)`}
   491→          />
   492→          <StatCard
   493→            icon={<AlertTriangle className="w-5 h-5 text-amber-600 dark:text-amber-400" />}
   494→            label="Outstanding"
   495→            value={fmtCurrency(outstanding)}
   496→            change={outstanding > 0 ? -5 : 0}
   497→            accent="bg-amber-100 dark:bg-amber-900/40"
   498→            sub={`${bills.filter(b => b.status !== 'Paid').length} unpaid bill(s)`}
   499→          />
   500→        </div>
   501→
   502→        {/* ── Additional stat row ────────────────────────────────────── */}
   503→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   504→          <StatCard
   505→            icon={<FileText className="w-5 h-5 text-blue-600 dark:text-blue-400" />}
   506→            label="Total Bills Generated"
   507→            value={fmt(bills.length)}
   508→            accent="bg-blue-100 dark:bg-blue-900/40"
   509→            sub={`Total billed: ${fmtCurrency(totalBilled)}`}
   510→          />
   511→          <StatCard
   512→            icon={<CheckCircle2 className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />}
   513→            label="Bills Paid"
   514→            value={fmt(bills.filter(b => b.status === 'Paid').length)}
   515→            accent="bg-emerald-100 dark:bg-emerald-900/40"
   516→            sub={`Value: ${fmtCurrency(totalPaidBills)}`}
   517→          />
   518→          <StatCard
   519→            icon={<Clock className="w-5 h-5 text-amber-600 dark:text-amber-400" />}
   520→            label="Bills Overdue"
   521→            value={fmt(bills.filter(b => b.status === 'Overdue').length)}
   522→            accent="bg-red-100 dark:bg-red-900/40"
   523→          />
   524→        </div>
   525→
   526→        {/* ── No data banner ──────────────────────────────────────────── */}
   527→        {!hasData && (
   528→          <div className="rounded-xl border-2 border-dashed border-slate-300 dark:border-slate-600 bg-white dark:bg-slate-800 p-8 text-center">
   529→            <BarChart3 className="w-12 h-12 text-slate-300 dark:text-slate-600 mx-auto mb-3" />
   530→            <h3 className="text-lg font-semibold text-slate-600 dark:text-slate-400 mb-2">No Data Yet</h3>
   531→            <p className="text-sm text-slate-400 dark:text-slate-500 max-w-md mx-auto">
   532→              Start by registering businesses, generating bills, or recording payments. 
   533→              Your dashboard analytics will populate automatically as data is entered into the system.
   534→            </p>
   535→          </div>
   536→        )}
   537→
   538→        {/* ── Charts Row 1 ────────────────────────────────────────────── */}
   539→        <div className="grid grid-cols-1 lg:grid-cols-5 gap-4">
   540→          {/* Monthly Revenue Trend — 3/5 width */}
   541→          <ChartCard title="Monthly Revenue Trend" className="lg:col-span-3">
   542→            {monthlyRevenue.length > 0 ? (
   543→              <div className="h-72 sm:h-80">
   544→                <ResponsiveContainer width="100%" height="100%">
