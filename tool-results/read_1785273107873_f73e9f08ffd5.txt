    99→export function PaymentsPage() {
   100→  const [payments, setPayments] = useState<Payment[]>(mockPayments);
   101→  const [search, setSearch] = useState('');
   102→  const [methodFilter, setMethodFilter] = useState<string>('All');
   103→  const [statusFilter, setStatusFilter] = useState<string>('All');
   104→  const [dateFrom, setDateFrom] = useState('');
   105→  const [dateTo, setDateTo] = useState('');
   106→
   107→  // Modal state
   108→  const [modalOpen, setModalOpen] = useState(false);
   109→  const [selectedBill, setSelectedBill] = useState('');
   110→  const [payAmount, setPayAmount] = useState('');
   111→  const [payMethod, setPayMethod] = useState<PaymentMethod>('Cash');
   112→  const [payReference, setPayReference] = useState('');
   113→  const [payCollector, setPayCollector] = useState('');
   114→  const [payRemarks, setPayRemarks] = useState('');
   115→
   116→  // Pagination
   117→  const [currentPage, setCurrentPage] = useState(1);
   118→  const perPage = 8;
   119→
   120→  const autoFill = useMemo(() => {
   121→    const bill = mockBills.find((b) => b.billNo === selectedBill);
   122→    if (!bill) return { business: '', balance: 0, totalAmount: 0 };
   123→    return { business: bill.business, balance: bill.balance, totalAmount: bill.totalAmount };
   124→  }, [selectedBill]);
   125→
   126→  // ─── Filtering ───────────────────────────────────────────────────────────
   127→
   128→  const filtered = useMemo(() => {
   129→    return payments.filter((p) => {
   130→      const matchSearch =
   131→        search === '' ||
   132→        p.receiptNo.toLowerCase().includes(search.toLowerCase()) ||
   133→        p.billNo.toLowerCase().includes(search.toLowerCase()) ||
   134→        p.business.toLowerCase().includes(search.toLowerCase()) ||
   135→        p.collector.toLowerCase().includes(search.toLowerCase());
   136→
   137→      const matchMethod = methodFilter === 'All' || p.method === methodFilter;
   138→      const matchStatus = statusFilter === 'All' || p.status === statusFilter;
   139→      const matchFrom = dateFrom === '' || p.date >= dateFrom;
   140→      const matchTo = dateTo === '' || p.date <= dateTo;
   141→
   142→      return matchSearch && matchMethod && matchStatus && matchFrom && matchTo;
   143→    });
   144→  }, [payments, search, methodFilter, statusFilter, dateFrom, dateTo]);
   145→
   146→  const totalPages = Math.ceil(filtered.length / perPage);
   147→  const paginated = filtered.slice((currentPage - 1) * perPage, currentPage * perPage);
   148→
   149→  // ─── Stats ──────────────────────────────────────────────────────────────
   150→
   151→  const totalPayments = payments.length;
   152→  const totalCollected = payments.reduce((sum, p) => sum + p.amount, 0);
   153→  const todayPayments = payments
   154→    .filter((p) => p.date === '2024-12-15')
   155→    .reduce((sum, p) => sum + p.amount, 0);
   156→  const totalPending = payments.reduce((sum, p) => sum + p.balance, 0);
   157→
   158→  // ─── Handlers ───────────────────────────────────────────────────────────
   159→
   160→  const openModal = () => {
   161→    setSelectedBill('');
   162→    setPayAmount('');
   163→    setPayMethod('Cash');
   164→    setPayReference('');
   165→    setPayCollector('');
   166→    setPayRemarks('');
   167→    setModalOpen(true);
   168→  };
   169→
   170→  const closeModal = () => setModalOpen(false);
   171→
   172→  const handleSave = () => {
   173→    if (!selectedBill || !payAmount) return;
   174→
   175→    const bill = mockBills.find((b) => b.billNo === selectedBill);
   176→    if (!bill) return;
   177→
   178→    const newPayment: Payment = {
   179→      id: String(Date.now()),
   180→      receiptNo: `RCP-2024-${String(payments.length + 1).padStart(4, '0')}`,
   181→      billNo: selectedBill,
   182→      business: bill.business,
   183→      amount: parseFloat(payAmount) || 0,
   184→      balance: parseFloat(payAmount) >= bill.totalAmount ? 0 : bill.balance - (parseFloat(payAmount) || 0),
   185→      date: new Date().toISOString().split('T')[0],
   186→      collector: payCollector || 'System',
   187→      method: payMethod,
   188→      status: parseFloat(payAmount) >= bill.totalAmount ? 'Full' : 'Partial',
   189→      reference: payReference,
   190→      remarks: payRemarks,
   191→    };
   192→
   193→    setPayments([newPayment, ...payments]);
   194→    closeModal();
   195→  };
   196→
   197→  const handleDelete = (id: string) => {
   198→    setPayments((prev) => prev.filter((p) => p.id !== id));
   199→  };
   200→
   201→  // ─── Render ─────────────────────────────────────────────────────────────
   202→
   203→  return (
   204→    <div className="space-y-6">
   205→      {/* ── Header ──────────────────────────────────────────────────────── */}
   206→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   207→        <div>
   208→          <h1 className="text-2xl font-bold text-gray-900">Payment Management</h1>
   209→          <p className="text-sm text-gray-500 mt-1">Record, track and manage all revenue payments</p>
   210→        </div>
   211→        <button
   212→          onClick={openModal}
   213→          className="inline-flex items-center gap-2 rounded-lg bg-emerald-600 px-5 py-2.5 text-sm font-semibold text-white shadow-sm hover:bg-emerald-700 transition-colors cursor-pointer"
   214→        >
   215→          <Plus className="h-4 w-4" />
   216→          Record Payment
   217→        </button>
   218→      </div>
   219→
   220→      {/* ── Stats ──────────────────────────────────────────────────────── */}
   221→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   222→        {[
   223→          { label: 'Total Payments', value: totalPayments.toLocaleString(), icon: Receipt, color: 'text-emerald-600', bg: 'bg-emerald-50' },
   224→          { label: 'Total Collected', value: formatCurrency(totalCollected), icon: DollarSign, color: 'text-blue-600', bg: 'bg-blue-50' },
   225→          { label: 'Today', value: formatCurrency(todayPayments), icon: CalendarCheck, color: 'text-amber-600', bg: 'bg-amber-50' },
   226→          { label: 'Pending Balance', value: formatCurrency(totalPending), icon: Clock, color: 'text-red-600', bg: 'bg-red-50' },
   227→        ].map((stat) => (
   228→          <div
   229→            key={stat.label}
   230→            className="flex items-center gap-4 rounded-lg border border-gray-200 bg-white p-4 shadow-sm"
   231→          >
   232→            <div className={`flex h-12 w-12 shrink-0 items-center justify-center rounded-lg ${stat.bg}`}>
   233→              <stat.icon className={`h-6 w-6 ${stat.color}`} />
   234→            </div>
   235→            <div>
   236→              <p className="text-xs font-medium text-gray-500 uppercase tracking-wider">{stat.label}</p>
   237→              <p className="text-lg font-bold text-gray-900">{stat.value}</p>
   238→            </div>
   239→          </div>
   240→        ))}
   241→      </div>
   242→
   243→      {/* ── Filters ─────────────────────────────────────────────────────── */}
   244→      <div className="rounded-lg border border-gray-200 bg-white p-4 shadow-sm">
   245→        <div className="flex items-center gap-2 mb-3">
   246→          <Filter className="h-4 w-4 text-gray-500" />
   247→          <span className="text-sm font-semibold text-gray-700">Filters</span>
   248→        </div>
   249→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-3">
   250→          {/* Search */}
   251→          <div className="relative">
   252→            <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   253→            <input
   254→              type="text"
   255→              placeholder="Search receipts, bills, business…"
   256→              value={search}
   257→              onChange={(e) => { setSearch(e.target.value); setCurrentPage(1); }}
   258→              className="w-full rounded-lg border border-gray-300 py-2 pl-10 pr-3 text-sm text-gray-900 placeholder-gray-400 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/20 focus:outline-none transition"
   259→            />
   260→          </div>
   261→          {/* Date From */}
   262→          <div className="relative">
   263→            <input
   264→              type="date"
   265→              value={dateFrom}
   266→              onChange={(e) => { setDateFrom(e.target.value); setCurrentPage(1); }}
   267→              className="w-full rounded-lg border border-gray-300 py-2 px-3 text-sm text-gray-900 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/20 focus:outline-none transition"
   268→            />
   269→          </div>
   270→          {/* Date To */}
   271→          <div className="relative">
   272→            <input
   273→              type="date"
   274→              value={dateTo}
   275→              onChange={(e) => { setDateTo(e.target.value); setCurrentPage(1); }}
   276→              className="w-full rounded-lg border border-gray-300 py-2 px-3 text-sm text-gray-900 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/20 focus:outline-none transition"
   277→            />
   278→          </div>
   279→          {/* Method */}
   280→          <div className="relative">
   281→            <select
   282→              value={methodFilter}
   283→              onChange={(e) => { setMethodFilter(e.target.value); setCurrentPage(1); }}
   284→              className="w-full appearance-none rounded-lg border border-gray-300 py-2 pl-3 pr-9 text-sm text-gray-900 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/20 focus:outline-none transition cursor-pointer"
   285→            >
   286→              <option value="All">All Methods</option>
   287→              <option value="Cash">Cash</option>
   288→              <option value="Mobile Money">Mobile Money</option>
   289→              <option value="Bank">Bank Transfer</option>
   290→              <option value="POS">POS</option>
   291→              <option value="Online">Online</option>
   292→            </select>
   293→            <ChevronDown className="pointer-events-none absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   294→          </div>
   295→          {/* Status */}
   296→          <div className="relative">
   297→            <select
   298→              value={statusFilter}
   299→              onChange={(e) => { setStatusFilter(e.target.value); setCurrentPage(1); }}
   300→              className="w-full appearance-none rounded-lg border border-gray-300 py-2 pl-3 pr-9 text-sm text-gray-900 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/20 focus:outline-none transition cursor-pointer"
   301→            >
   302→              <option value="All">All Statuses</option>
   303→              <option value="Full">Full</option>
   304→              <option value="Partial">Partial</option>
   305→              <option value="Advance">Advance</option>
   306→            </select>
   307→            <ChevronDown className="pointer-events-none absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 text-gray-400" />
   308→          </div>
   309→        </div>
   310→      </div>
   311→
   312→      {/* ── Table ───────────────────────────────────────────────────────── */}
   313→      <div className="rounded-lg border border-gray-200 bg-white shadow-sm overflow-hidden">
   314→        <div className="overflow-x-auto">
   315→          <table className="w-full min-w-[1100px] text-sm">
   316→            <thead>
   317→              <tr className="border-b border-gray-200 bg-gray-50">
   318→                <th className="px-4 py-3 text-left font-semibold text-gray-600">Receipt #</th>
   319→                <th className="px-4 py-3 text-left font-semibold text-gray-600">Bill #</th>
   320→                <th className="px-4 py-3 text-left font-semibold text-gray-600">Business</th>
   321→                <th className="px-4 py-3 text-right font-semibold text-gray-600">Amount</th>
   322→                <th className="px-4 py-3 text-right font-semibold text-gray-600">Balance</th>
   323→                <th className="px-4 py-3 text-left font-semibold text-gray-600">Date</th>
   324→                <th className="px-4 py-3 text-left font-semibold text-gray-600">Collector</th>
   325→                <th className="px-4 py-3 text-center font-semibold text-gray-600">Method</th>
   326→                <th className="px-4 py-3 text-center font-semibold text-gray-600">Status</th>
   327→                <th className="px-4 py-3 text-center font-semibold text-gray-600">Actions</th>
   328→              </tr>
   329→            </thead>
   330→            <tbody className="divide-y divide-gray-100">
   331→              {paginated.length === 0 ? (
   332→                <tr>
   333→                  <td colSpan={10} className="px-4 py-12 text-center text-gray-400">
   334→                    No payments found matching your filters.
   335→                  </td>
   336→                </tr>
   337→              ) : (
   338→                paginated.map((p) => {
   339→                  const m = methodBadge[p.method];
   340→                  const s = statusBadge[p.status];
   341→                  return (
   342→                    <tr
   343→                      key={p.id}
   344→                      className="hover:bg-gray-50/60 transition-colors"
   345→                    >
   346→                      <td className="px-4 py-3 font-medium text-emerald-700">{p.receiptNo}</td>
   347→                      <td className="px-4 py-3 text-gray-600">{p.billNo}</td>
   348→                      <td className="px-4 py-3 text-gray-900 font-medium max-w-[200px] truncate">{p.business}</td>
   349→                      <td className="px-4 py-3 text-right text-gray-900 font-semibold whitespace-nowrap">{formatCurrency(p.amount)}</td>

   353→                      <td className="px-4 py-3 text-gray-500 whitespace-nowrap">{p.date}</td>

   355→                      <td className="px-4 py-3 text-center">
   356→                        <span className={`inline-flex items-center rounded-full px-2.5 py-0.5 text-xs font-semibold ${m.bg}`}>
   357→                          {m.text}
   358→                        </span>
   359→                      </td>
   360→                      <td className="px-4 py-3 text-center">
   361→                        <span className={`inline-flex items-center gap-1.5 rounded-full px-2.5 py-0.5 text-xs font-semibold ${s.bg}`}>
   362→                          <span className={`h-1.5 w-1.5 rounded-full ${s.dot}`} />
   363→                          {s.text}
   364→                        </span>
   365→                      </td>
   366→                      <td className="px-4 py-3">
   367→                        <div className="flex items-center justify-center gap-1">
