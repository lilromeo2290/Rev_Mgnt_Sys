   382→                      </span>
   383→                    </td>
   384→                    <td className="px-4 py-3">
   385→                      <div className="flex items-center justify-center gap-1">
   386→                        <button
   387→                          onClick={() => setSelectedReceipt(r)}
   388→                          className="p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   389→                          title="View details"
   390→                        >
   391→                          <Eye className="w-4 h-4" />
   392→                        </button>
   393→                        <button
   394→                          onClick={() => handlePrintReceipt(r)}
   395→                          className="p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   396→                          title="Print receipt"
   397→                        >
   398→                          <Printer className="w-4 h-4" />
   399→                        </button>
   400→                      </div>
   401→                    </td>
   402→                  </tr>
   403→                ))
   404→              )}
   405→            </tbody>
   406→          </table>
   407→        </div>
   408→
   409→        {/* Pagination */}
   410→        <div className="flex items-center justify-between px-4 py-3 border-t border-slate-200 dark:border-slate-700">
   411→          <p className="text-sm text-slate-500 dark:text-slate-400">
   412→            Showing {paginated.length} of {filtered.length} receipts
   413→          </p>
   414→          <div className="flex items-center gap-1">
   415→            <button
   416→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   417→              disabled={page === 1}
   418→              className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   419→            >
   420→              <ChevronLeft className="w-4 h-4" />
   421→            </button>
   422→            {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
   423→              <button
   424→                key={p}
   425→                onClick={() => setPage(p)}
   426→                className={`w-8 h-8 rounded-lg text-sm font-medium transition-colors ${
   427→                  p === page
   428→                    ? 'bg-emerald-600 text-white'
   429→                    : 'text-slate-500 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-700'
   430→                }`}
   431→              >
   432→                {p}
   433→              </button>
   434→            ))}
   435→            <button
   436→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   437→              disabled={page === totalPages}
   438→              className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   439→            >
   440→              <ChevronRight className="w-4 h-4" />
   441→            </button>
   442→          </div>
   443→        </div>
   444→      </div>
   445→
   446→      {/* Detail Modal */}
   447→      {selectedReceipt && (
   448→        <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
   449→          <div className="fixed inset-0 bg-black/50 backdrop-blur-sm" onClick={() => setSelectedReceipt(null)} />
   450→          <div className="relative bg-white dark:bg-slate-800 rounded-xl border border-slate-200 dark:border-slate-700 shadow-2xl w-full max-w-lg max-h-[90vh] overflow-y-auto">
   451→            {/* Modal Header */}
   452→            <div className="flex items-center justify-between p-5 border-b border-slate-200 dark:border-slate-700">
   453→              <div className="flex items-center gap-3">
   454→                <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   455→                  <Receipt className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   456→                </div>
   457→                <div>
   458→                  <h2 className="text-lg font-semibold text-slate-900 dark:text-white">Receipt Details</h2>
   459→                  <p className="text-sm font-mono text-slate-500 dark:text-slate-400">{selectedReceipt.receiptNo}</p>
   460→                </div>
   461→              </div>
   462→              <button
   463→                onClick={() => setSelectedReceipt(null)}
   464→                className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 transition-colors"
   465→              >
   466→                <X className="w-5 h-5" />
   467→              </button>
   468→            </div>
   469→
   470→            {/* Modal Body */}
   471→            <div className="p-5 space-y-5">
   472→              {/* Status Badge */}
   473→              <div className="flex items-center justify-between">
   474→                <span className={`inline-flex items-center gap-1.5 text-xs font-semibold px-2.5 py-1 rounded-full ${statusStyle[selectedReceipt.status].pill}`}>
   475→                  {selectedReceipt.status}
   476→                </span>
   477→                <div className="flex items-center gap-2">
   478→                  <button onClick={() => handlePrintReceipt(selectedReceipt)} className="inline-flex items-center gap-1.5 rounded-lg bg-emerald-600 text-white px-3 py-1.5 text-sm font-medium hover:bg-emerald-700 transition-colors cursor-pointer">
   479→                    <Printer className="w-3.5 h-3.5" />
   480→                    Print
   481→                  </button>
   482→                  <button onClick={() => handlePrintReceipt(selectedReceipt)} className="inline-flex items-center gap-1.5 rounded-lg border border-slate-200 dark:border-slate-700 text-slate-700 dark:text-slate-300 px-3 py-1.5 text-sm font-medium hover:bg-slate-50 dark:hover:bg-slate-700 transition-colors cursor-pointer">
   483→                    <Download className="w-3.5 h-3.5" />
   484→                    PDF
   485→                  </button>
   486→                </div>
   487→              </div>
   488→
   489→              {/* Info Grid */}
   490→              <div className="grid grid-cols-2 gap-4">
   491→                <InfoField label="Bill Reference" value={selectedReceipt.billNo} />
   492→                <InfoField label="Payment Method" value={selectedReceipt.method} />
   493→                <InfoField label="Transaction Ref" value={selectedReceipt.reference} />
   494→                <InfoField label="Zone / Ward" value={`${selectedReceipt.zone} — ${selectedReceipt.ward}`} />
   495→                <InfoField label="Collected By" value={selectedReceipt.issuedBy} />
   496→                <InfoField label="Entity Type" value={selectedReceipt.entityType} />
   497→              </div>
   498→
   499→              {/* Issued To */}
   500→              <div className="rounded-lg bg-slate-50 dark:bg-slate-900/50 p-4">
   501→                <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium mb-1">Issued To</p>
   502→                <p className="text-lg font-semibold text-slate-900 dark:text-white">{selectedReceipt.issuedTo}</p>
   503→              </div>
   504→
   505→              {/* Line Items */}
   506→              <div>
   507→                <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium mb-3">Line Items</p>
   508→                <div className="rounded-lg border border-slate-200 dark:border-slate-700 overflow-hidden">
   509→                  <table className="w-full text-left text-sm">
   510→                    <thead className="bg-slate-50 dark:bg-slate-900/50">
   511→                      <tr className="border-b border-slate-200 dark:border-slate-700">
   512→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400">Description</th>
   513→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Qty</th>
   514→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Unit Price</th>
   515→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Amount</th>
   516→                      </tr>
   517→                    </thead>
   518→                    <tbody className="divide-y divide-slate-100 dark:divide-slate-700/60">
   519→                      {selectedReceipt.items.map((item, idx) => (
   520→                        <tr key={idx}>
   521→                          <td className="px-3 py-2 text-slate-700 dark:text-slate-300">{item.description}</td>
   522→                          <td className="px-3 py-2 text-slate-600 dark:text-slate-400 text-right">{item.quantity}</td>
   523→                          <td className="px-3 py-2 text-slate-600 dark:text-slate-400 text-right">{fmtCurrency(item.unitPrice)}</td>
   524→                          <td className="px-3 py-2 text-slate-900 dark:text-white font-medium text-right">{fmtCurrency(item.amount)}</td>
   525→                        </tr>
   526→                      ))}
   527→                    </tbody>
   528→                    <tfoot className="bg-slate-50 dark:bg-slate-900/50">
   529→                      <tr className="border-t border-slate-200 dark:border-slate-700">
   530→                        <td colSpan={3} className="px-3 py-2.5 text-sm font-semibold text-slate-700 dark:text-slate-300 text-right">Total Paid</td>
   531→                        <td className="px-3 py-2.5 text-sm font-bold text-emerald-600 dark:text-emerald-400 text-right">{fmtCurrency(selectedReceipt.totalPaid)}</td>
   532→                      </tr>
   533→                    </tfoot>
   534→                  </table>
   535→                </div>
   536→              </div>
   537→
   538→              {/* Date / Time */}
   539→              <div className="flex items-center gap-6 text-sm text-slate-500 dark:text-slate-400">
   540→                <span className="flex items-center gap-1.5"><CalendarCheck className="w-4 h-4" />{selectedReceipt.date}</span>
   541→                <span className="flex items-center gap-1.5"><UserCheck className="w-4 h-4" />{selectedReceipt.issuedBy}</span>
   542→              </div>
   543→            </div>
   544→          </div>
   545→        </div>
   546→      )}
   547→    </div>
   548→  );
   549→}
