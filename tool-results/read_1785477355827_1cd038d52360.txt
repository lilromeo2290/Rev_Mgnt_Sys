    87→function loadEntities(): { id: string; name: string; type: string; category?: string }[] {
    88→  try {
    89→    const businesses = JSON.parse(localStorage.getItem('rms-businesses') || '[]') as { regNumber: string; name: string; type: string; category: string }[];
    90→    const properties = JSON.parse(localStorage.getItem('rms-properties') || '[]') as { propNumber: string; ownerName: string; propertyUseType: string; category: string }[];
    91→    const rents = JSON.parse(localStorage.getItem('rms-rents') || '[]') as { id: string; rentObjectName: string; rentClass: string; rentCategory: string }[];
    92→
    93→    const list: { id: string; name: string; type: string; category?: string }[] = [];
    94→
    95→    for (const b of businesses) {
    96→      if (b.name) list.push({ id: b.regNumber || String(Date.now()), name: b.name, type: 'Business', category: b.category || b.type || '' });
    97→    }
    98→    for (const p of properties) {
    99→      if (p.ownerName) list.push({ id: p.propNumber || String(Date.now()), name: p.ownerName, type: 'Property', category: p.category || p.propertyUseType || '' });
   100→    }
   101→    for (const r of rents) {
   102→      if (r.rentObjectName) list.push({ id: r.id || String(Date.now()), name: r.rentObjectName, type: 'Rent', category: r.rentClass || r.rentCategory || '' });
   103→    }
   104→
   105→    return list;
   106→  } catch {
   107→    return [];
   108→  }
   109→}
