   338→   338→  const [statusFilter, setStatusFilter] = useState<string>('All');
   339→   339→  const [categoryFilter, setCategoryFilter] = useState<string>('All');
   340→   340→  const [revenueAreaFilter, setRevenueAreaFilter] = useState<string>('All');
   341→   341→  const [dateFrom, setDateFrom] = useState('');

   344→   344→  const [showModal, setShowModal] = useState(false);

   349→   349→  const [formData, setFormData] = useState<BillFormData>({
   350→   350→    entityName: '',
   351→   351→    entityType: 'Business',
   352→   352→    revenueItem: '',

   361→   361→  const filtered = useMemo(() => {
   362→   362→    return bills.filter((b) => {
   363→   363→      const matchSearch =
   364→   364→        searchQuery === '' ||
   365→   365→        b.entityName.toLowerCase().includes(searchQuery.toLowerCase()) ||
   366→   366→        b.billNumber.toLowerCase().includes(searchQuery.toLowerCase()) ||
   367→   367→        b.revenueItem.toLowerCase().includes(searchQuery.toLowerCase());
   368→   368→      const matchStatus =
   369→   369→        statusFilter === 'All' || b.status === statusFilter;
   370→   370→      const matchCategory =
   371→   371→        categoryFilter === 'All' || b.category === categoryFilter;
   372→   372→      const matchRevenueArea =
   373→   373→        revenueAreaFilter === 'All' ||
   374→   374→        (b.entityType === revenueAreaFilter);
   375→   375→      const matchDateFrom =
   376→   376→        dateFrom === '' || b.date >= dateFrom;
   377→   377→      const matchDateTo =
   378→   378→        dateTo === '' || b.date <= dateTo;
   379→   379→      return matchSearch && matchStatus && matchCategory && matchRevenueArea && matchDateFrom && matchDateTo;
   380→   380→    });
   381→   381→  }, [bills, searchQuery, statusFilter, categoryFilter, revenueAreaFilter, dateFrom, dateTo]);
   382→   382→
   383→   383→  // ── Pagination ──────────────────────────────────────────────────────────
   384→   384→

   386→   386→  const safeCurrentPage = Math.min(currentPage, totalPages);

   388→   388→  const paged = filtered.slice(startIdx, startIdx + itemsPerPage);

   401→   401→    return (
   402→   402→      <span className={`inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-xs font-semibold ${styles[status]}`}>

   415→   415→
   416→   416→  const stats = useMemo(() => {
   417→   417→    const totalBilled = bills.reduce((sum, b) => sum + b.totalDue, 0);
   418→   418→    const paid = bills
   419→   419→      .filter((b) => b.status === 'Paid')
   420→   420→      .reduce((sum, b) => sum + b.totalDue, 0);
   421→   421→    const outstanding = totalBilled - paid;
   422→   422→    const overdue = bills
   423→   423→      .filter((b) => b.status === 'Overdue')
   424→   424→      .reduce((sum, b) => sum + b.totalDue, 0);
   425→   425→    return {
   426→   426→      total: bills.length,
   427→   427→      totalBilled,
   428→   428→      outstanding,

   430→   430→    };
   431→   431→  }, [bills]);

   434→   434→
   435→   435→  const handleRevenueItemChange = (item: string) => {

   438→   438→    setFormData((p) => ({

   440→   440→      revenueItem: item,

   444→   444→  };

   450→   450→

   458→   458→      entityName: formData.entityName,
   459→   459→      entityType: formData.entityType,
   460→   460→      category: ENTITIES.find((e) => e.name === formData.entityName)?.category ?? 'General',
   461→   461→      revenueItem: formData.revenueItem,
   462→   462→      amount: formData.amount,
   463→   463→      previousBalance: formData.previousBalance,
   464→   464→      penalty: formData.penalty,
   465→   465→      totalDue,
   466→   466→      status: 'Unpaid',
   467→   467→      dueDate: formData.dueDate || '',
   468→   468→    };
   469→   469→
   470→   470→    setBills((prev) => [newBill, ...prev]);
   471→   471→    setShowModal(false);
   472→   472→    setFormData({
   473→   473→      entityName: '',
   474→   474→      entityType: 'Business',
   475→   475→      revenueItem: '',
   476→   476→      amount: 0,
   477→   477→      previousBalance: 0,
   478→   478→      penalty: 0,
   479→   479→      dueDate: '',
   480→   480→    });
   481→   481→    setCurrentPage(1);
