   115→const defaultUsers: User[] = [

   118→
   119→function loadUsers(): User[] {
   120→  if (typeof window === 'undefined') return defaultUsers;
   121→  try {
   122→    const stored = localStorage.getItem(USERS_STORAGE_KEY);
   123→    if (stored) {
   124→      const parsed = JSON.parse(stored);
   125→      if (Array.isArray(parsed) && parsed.length > 0) return parsed;
   126→    }
   127→  } catch { /* ignore */ }
   128→  return defaultUsers;
   129→}
   130→
   131→function saveUsers(users: User[]) {
   132→  if (typeof window === 'undefined') return;
   133→  try { localStorage.setItem(USERS_STORAGE_KEY, JSON.stringify(users)); } catch { /* ignore */ }
   134→}
