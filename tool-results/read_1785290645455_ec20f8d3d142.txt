   335→// ─── Component ───────────────────────────────────────────────────────────────
   336→
   337→export function BillingPage() {
   338→  const [bills, setBills] = useState<Bill[]>(initialBills);
   339→  const [searchQuery, setSearchQuery] = useState('');
   340→  const [statusFilter, setStatusFilter] = useState<string>('All');
   341→  const [categoryFilter, setCategoryFilter] = useState<string>('All');
   342→  const [revenueAreaFilter, setRevenueAreaFilter] = useState<string>('All');
   343→  const [dateFrom, setDateFrom] = useState('');
   344→  const [dateTo, setDateTo] = useState('');
   345→  const [currentPage, setCurrentPage] = useState(1);
   346→  const [showModal, setShowModal] = useState(false);
   347→  const [showBulkModal, setShowBulkModal] = useState(false);
   348→  const [bulkForm, setBulkForm] = useState({
   349→    entityType: 'All' as 'All' | 'Business' | 'Property',
   350→    category: 'All',
   351→    revenueItem: '',
   352→    dueDate: '',
   353→  });
   354→  const [viewingBill, setViewingBill] = useState<Bill | null>(null);
   355→  const [bulkProgress, setBulkProgress] = useState<'idle' | 'generating' | 'done'>('idle');
   356→  const [bulkGeneratedCount, setBulkGeneratedCount] = useState(0);
   357→  const itemsPerPage = 8;
   358→
   359→  // ── Form data ───────────────────────────────────────────────────────────
   360→
   361→  const [formData, setFormData] = useState<BillFormData>({
   362→    entityName: '',
   363→    entityType: 'Business',
   364→    revenueItem: '',
   365→    amount: 0,
   366→    previousBalance: 0,
   367→    penalty: 0,
   368→    dueDate: '',
   369→  });
   370→
   371→  // ── Filtering ───────────────────────────────────────────────────────────
   372→
   373→  const filtered = useMemo(() => {
   374→    return bills.filter((b) => {
   375→      const matchSearch =
   376→        searchQuery === '' ||
   377→        b.entityName.toLowerCase().includes(searchQuery.toLowerCase()) ||
   378→        b.billNumber.toLowerCase().includes(searchQuery.toLowerCase()) ||
   379→        b.revenueItem.toLowerCase().includes(searchQuery.toLowerCase());
   380→      const matchStatus =
   381→        statusFilter === 'All' || b.status === statusFilter;
   382→      const matchCategory =
   383→        categoryFilter === 'All' || b.category === categoryFilter;
   384→      const matchRevenueArea =
   385→        revenueAreaFilter === 'All' ||
   386→        (b.entityType === revenueAreaFilter);
   387→      const matchDateFrom =
   388→        dateFrom === '' || b.date >= dateFrom;
   389→      const matchDateTo =
   390→        dateTo === '' || b.date <= dateTo;
   391→      return matchSearch && matchStatus && matchCategory && matchRevenueArea && matchDateFrom && matchDateTo;
   392→    });
   393→  }, [bills, searchQuery, statusFilter, categoryFilter, revenueAreaFilter, dateFrom, dateTo]);
   394→
   395→  // ── Pagination ──────────────────────────────────────────────────────────
   396→
   397→  const totalPages = Math.max(1, Math.ceil(filtered.length / itemsPerPage));
   398→  const safeCurrentPage = Math.min(currentPage, totalPages);
   399→  const startIdx = (safeCurrentPage - 1) * itemsPerPage;
   400→  const paged = filtered.slice(startIdx, startIdx + itemsPerPage);
   401→  const showingFrom = filtered.length === 0 ? 0 : startIdx + 1;
   402→  const showingTo = Math.min(startIdx + itemsPerPage, filtered.length);
   403→
   404→  // ── Status badge ───────────────────────────────────────────────────────
   405→
   406→  const StatusBadge = ({ status }: { status: Bill['status'] }) => {
   407→    const styles: Record<Bill['status'], string> = {
   408→      Paid: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400',
   409→      Partial: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400',
   410→      Unpaid: 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400',
   411→      Overdue: 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400',
   412→    };
   413→    return (
   414→      <span className={`inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-xs font-semibold ${styles[status]}`}>
   415→        {status === 'Overdue' && (
   416→          <span className="relative flex h-2 w-2">
   417→            <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-red-400 opacity-75" />
   418→            <span className="relative inline-flex rounded-full h-2 w-2 bg-red-500" />
   419→          </span>
   420→        )}
   421→        {status}
   422→      </span>
   423→    );
   424→  };
   425→
   426→  // ── Stats ──────────────────────────────────────────────────────────────
   427→
   428→  const stats = useMemo(() => {
   429→    const totalBilled = bills.reduce((sum, b) => sum + b.totalDue, 0);
   430→    const paid = bills
   431→      .filter((b) => b.status === 'Paid')
   432→      .reduce((sum, b) => sum + b.totalDue, 0);
   433→    const outstanding = totalBilled - paid;
   434→    const overdue = bills
   435→      .filter((b) => b.status === 'Overdue')
   436→      .reduce((sum, b) => sum + b.totalDue, 0);
   437→    return {
   438→      total: bills.length,
   439→      totalBilled,
   440→      outstanding,
   441→      overdue,
   442→    };
   443→  }, [bills]);
   444→
   445→  // ── Auto-fill amount when revenue item changes ──────────────────────────
   446→
   447→  const handleRevenueItemChange = (item: string) => {
   448→    const amount = RATE_AMOUNTS[item] || 0;
   449→    const penalty = Math.round(amount * 0.05 * 100) / 100;
   450→    setFormData((p) => ({
   451→      ...p,
   452→      revenueItem: item,
   453→      amount,
   454→      penalty,
   455→    }));
   456→  };
   457→
   458→  // ── Generate bill ──────────────────────────────────────────────────────
   459→
   460→  const handleGenerateBill = () => {
   461→    if (!formData.entityName || !formData.revenueItem || formData.amount <= 0) return;
   462→
   463→    const newBillNumber = `BILL-2024-${String(bills.length + 156).padStart(4, '0')}`;
   464→    const totalDue = formData.amount + formData.previousBalance + formData.penalty;
   465→
   466→    const newBill: Bill = {
   467→      id: String(bills.length + 1),
   468→      billNumber: newBillNumber,
   469→      date: new Date().toISOString().split('T')[0],
   470→      entityName: formData.entityName,
   471→      entityType: formData.entityType,
   472→      category: ENTITIES.find((e) => e.name === formData.entityName)?.category ?? 'General',
   473→      revenueItem: formData.revenueItem,
   474→      amount: formData.amount,
   475→      previousBalance: formData.previousBalance,
   476→      penalty: formData.penalty,
   477→      totalDue,
   478→      status: 'Unpaid',
   479→      dueDate: formData.dueDate || '',
   480→    };
   481→
   482→    setBills((prev) => [newBill, ...prev]);
   483→    setShowModal(false);
   484→    setFormData({
   485→      entityName: '',
   486→      entityType: 'Business',
   487→      revenueItem: '',
   488→      amount: 0,
   489→      previousBalance: 0,
   490→      penalty: 0,
   491→      dueDate: '',
   492→    });
   493→    setCurrentPage(1);
   494→  };
   495→
   496→  // ── Cancel bill ────────────────────────────────────────────────────────
   497→
   498→  const handleCancelBill = (id: string) => {
   499→    setBills((prev) => prev.filter((b) => b.id !== id));
   500→  };
   501→
   502→  // ── Bulk eligible count ───────────────────────────────────────────────
   503→
   504→  const bulkEligibleCount = useMemo(() => {
   505→    if (!bulkForm.revenueItem) return 0;
   506→    return ENTITIES.filter((e) => {
   507→      if (bulkForm.entityType !== 'All' && e.type !== bulkForm.entityType) return false;
   508→      if (bulkForm.category !== 'All' && e.category !== bulkForm.category) return false;
   509→      const exists = bills.find(
   510→        (b) => b.entityName === e.name && b.revenueItem === bulkForm.revenueItem,
   511→      );
   512→      return !exists;
   513→    }).length;
   514→  }, [bulkForm, bills]);
   515→
   516→  // ── Bulk generate bills ────────────────────────────────────────────────
   517→
   518→  const handleBulkGenerate = () => {
   519→    if (!bulkForm.revenueItem || bulkEligibleCount === 0) return;
   520→
   521→    setBulkProgress('generating');
   522→
   523→    const amount = RATE_AMOUNTS[bulkForm.revenueItem] || 0;
   524→    const penalty = Math.round(amount * 0.05 * 100) / 100;
   525→
   526→    setTimeout(() => {
   527→      const eligibleEntities = ENTITIES.filter((e) => {
   528→        if (bulkForm.entityType !== 'All' && e.type !== bulkForm.entityType) return false;
   529→        if (bulkForm.category !== 'All' && e.category !== bulkForm.category) return false;
   530→        const exists = bills.find(
   531→          (b) => b.entityName === e.name && b.revenueItem === bulkForm.revenueItem,
   532→        );
   533→        return !exists;
   534→      });
   535→
   536→      const newBills: Bill[] = eligibleEntities.map((entity, idx) => {
   537→        const totalDue = amount + penalty;
   538→        return {
   539→          id: String(Date.now() + idx),
   540→          billNumber: `BILL-2024-${String(bills.length + 156 + idx).padStart(4, '0')}`,
   541→          date: new Date().toISOString().split('T')[0],
   542→          entityName: entity.name,
   543→          entityType: entity.type,
   544→          category: entity.category,
   545→          revenueItem: bulkForm.revenueItem,
   546→          amount,
   547→          previousBalance: 0,
   548→          penalty,
   549→          totalDue,
   550→          status: 'Unpaid' as const,
   551→          dueDate: bulkForm.dueDate || new Date(Date.now() + 30 * 86400000).toISOString().split('T')[0],
   552→        };
   553→      });
   554→
   555→      setBills((prev) => [...newBills.reverse(), ...prev]);
   556→      setBulkGeneratedCount(newBills.length);
   557→      setBulkProgress('done');
   558→      setCurrentPage(1);
   559→    }, 800);
   560→  };
   561→
   562→  const handleCloseBulkModal = () => {
   563→    setShowBulkModal(false);
   564→    setBulkForm({ entityType: 'All', category: 'All', revenueItem: '', dueDate: '' });
   565→    setBulkProgress('idle');
   566→    setBulkGeneratedCount(0);
   567→  };
   568→
   569→  // ── View bill ──────────────────────────────────────────────────────────
   570→
   571→  const handleViewBill = (bill: Bill) => {
   572→    setViewingBill(bill);
   573→  };
   574→
   575→  // ── Print bill ─────────────────────────────────────────────────────────
   576→
