   111→
   112→export function UsersPage() {
   113→  const [users, setUsers] = useState<User[]>(initialUsers);
   114→  const [search, setSearch] = useState('');
   115→  const [roleFilter, setRoleFilter] = useState<'All' | UserRole>('All');
   116→  const [statusFilter, setStatusFilter] = useState<'All' | UserStatus>('All');
   117→  const [page, setPage] = useState(1);
   118→  const [showAddModal, setShowAddModal] = useState(false);
   119→  const [editUser, setEditUser] = useState<User | null>(null);
   120→  const [form, setForm] = useState<UserFormData>(emptyForm);
   121→  const [showFilters, setShowFilters] = useState(false);
   122→  const [viewUser, setViewUser] = useState<User | null>(null);
   123→
   124→  const filtered = useMemo(() => {
   125→    return users.filter((u) => {
   126→      const q = search.toLowerCase();
   127→      const fullName = `${u.firstName} ${u.lastName}`.toLowerCase();
   128→      const matchSearch =
   129→        !q ||
   130→        fullName.includes(q) ||
   131→        u.staffId.toLowerCase().includes(q) ||
   132→        u.email.toLowerCase().includes(q) ||
   133→        u.zone.toLowerCase().includes(q);
   134→      const matchRole = roleFilter === 'All' || u.role === roleFilter;
   135→      const matchStatus = statusFilter === 'All' || u.status === statusFilter;
   136→      return matchSearch && matchRole && matchStatus;
   137→    });
   138→  }, [users, search, roleFilter, statusFilter]);
   139→

   141→  const paginated = filtered.slice((page - 1) * ITEMS_PER_PAGE, page * ITEMS_PER_PAGE);
   142→
   143→  const activeCount = users.filter((u) => u.status === 'Active').length;
   144→  const inactiveCount = users.filter((u) => u.status === 'Inactive').length;
   145→  const suspendedCount = users.filter((u) => u.status === 'Suspended').length;
   146→

   148→    setForm(emptyForm);
   149→    setShowAddModal(true);
   150→  };

   153→    setForm({
   154→      firstName: user.firstName,
   155→      lastName: user.lastName,
   156→      email: user.email,
   157→      phone: user.phone,
   158→      role: user.role,
   159→      zone: user.zone,
   160→      ward: user.ward,
   161→      ghanaCard: user.ghanaCard,
   162→      accessiblePages: user.accessiblePages || ROLE_DEFAULT_PAGES[user.role],
   163→    });
   164→    setEditUser(user);
   165→  };
   166→
   167→  // When role changes, auto-set default pages (admin can still customize)
   168→  const handleRoleChange = (role: UserRole) => {
   169→    setForm((prev) => ({
   170→      ...prev,
   171→      role,
   172→      accessiblePages: ROLE_DEFAULT_PAGES[role],
   173→    }));
   174→  };
   175→
   176→  const togglePage = (page: RMSPage) => {
   177→    setForm((prev) => ({
   178→      ...prev,
   179→      accessiblePages: prev.accessiblePages.includes(page)
   180→        ? prev.accessiblePages.filter((p) => p !== page)
   181→        : [...prev.accessiblePages, page],
   182→    }));
   183→  };
   184→
   185→  const selectAllPages = () => setForm((prev) => ({ ...prev, accessiblePages: [...ALL_PAGES] }));
   186→  const deselectAllPages = () => setForm((prev) => ({ ...prev, accessiblePages: [] }));
   187→
   188→  const handleSave = () => {
   189→    if (editUser) {
   190→      setUsers((prev) =>
   191→        prev.map((u) =>
   192→          u.id === editUser.id
   193→            ? { ...u, ...form }
   194→            : u
   195→        )
   196→      );
   197→      setEditUser(null);
   198→    } else {
   199→      const newUser: User = {
   200→        id: String(Date.now()),
   201→        staffId: `STF-${String(users.length + 1).padStart(3, '0')}`,
   202→        ...form,
   203→        status: 'Active',
   204→        lastLogin: 'Never',
   205→        dateCreated: new Date().toISOString().slice(0, 10),
   206→      };
   207→      setUsers((prev) => [newUser, ...prev]);
   208→      setShowAddModal(false);
   209→    }
   210→    setForm(emptyForm);
   211→  };
   212→
   213→  const handleDelete = (id: string) => {
   214→    setUsers((prev) => prev.filter((u) => u.id !== id));
   215→  };
   216→
   217→  const toggleStatus = (id: string) => {
   218→    setUsers((prev) =>
   219→      prev.map((u) =>
   220→        u.id === id
   221→          ? { ...u, status: u.status === 'Active' ? 'Inactive' : 'Active' }
   222→          : u
   223→      )
   224→    );
   225→  };
   226→
   227→  const updateForm = (field: keyof UserFormData, value: string) => {
   228→    setForm((prev) => ({ ...prev, [field]: value }));
   229→  };
   230→
   231→  const isValid =
   232→    form.firstName.trim() !== '' &&
   233→    form.lastName.trim() !== '' &&
   234→    form.email.trim() !== '' &&
   235→    form.phone.trim() !== '';
   236→
   237→  return (
   238→    <div className="space-y-6">
   239→      {/* Header */}
   240→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   241→        <div>
   242→          <h1 className="text-2xl font-bold text-slate-900 dark:text-white">User Management</h1>
   243→          <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">
   244→            Manage system users, roles, and access permissions
   245→          </p>
   246→        </div>
   247→        <button
   248→          onClick={openAdd}
   249→          className="inline-flex items-center gap-2 rounded-lg bg-emerald-600 hover:bg-emerald-700 text-white px-4 py-2 text-sm font-medium transition-colors"
   250→        >
   251→          <Plus className="w-4 h-4" />
   252→          Add User
   253→        </button>
   254→      </div>
   255→
   256→      {/* Summary Cards */}
   257→      <div className="grid grid-cols-1 sm:grid-cols-4 gap-4">
   258→        <div className="rounded-xl bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 p-4 flex items-center gap-4">
   259→          <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   260→            <Users className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   261→          </div>
   262→          <div>
   263→            <p className="text-sm text-slate-500 dark:text-slate-400">Total Users</p>
   264→            <p className="text-xl font-bold text-slate-900 dark:text-white">{users.length}</p>
   265→          </div>
   266→        </div>
   267→        <div className="rounded-xl bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 p-4 flex items-center gap-4">
   268→          <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   269→            <CheckCircle2 className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   270→          </div>
   271→          <div>
   272→            <p className="text-sm text-slate-500 dark:text-slate-400">Active</p>
   273→            <p className="text-xl font-bold text-slate-900 dark:text-white">{activeCount}</p>
   274→          </div>
   275→        </div>
   276→        <div className="rounded-xl bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 p-4 flex items-center gap-4">
   277→          <div className="w-10 h-10 rounded-lg bg-slate-100 dark:bg-slate-700 flex items-center justify-center">
   278→            <Clock className="w-5 h-5 text-slate-500 dark:text-slate-400" />
   279→          </div>
   280→          <div>
   281→            <p className="text-sm text-slate-500 dark:text-slate-400">Inactive</p>
   282→            <p className="text-xl font-bold text-slate-900 dark:text-white">{inactiveCount}</p>
   283→          </div>
   284→        </div>
   285→        <div className="rounded-xl bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 p-4 flex items-center gap-4">
   286→          <div className="w-10 h-10 rounded-lg bg-red-100 dark:bg-red-900/30 flex items-center justify-center">
   287→            <XCircle className="w-5 h-5 text-red-500" />
   288→          </div>
   289→          <div>
   290→            <p className="text-sm text-slate-500 dark:text-slate-400">Suspended</p>
   291→            <p className="text-xl font-bold text-slate-900 dark:text-white">{suspendedCount}</p>
   292→          </div>
   293→        </div>
   294→      </div>
   295→
   296→      {/* Table Card */}
   297→      <div className="rounded-xl bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 overflow-hidden">
   298→        {/* Toolbar */}
   299→        <div className="p-4 border-b border-slate-200 dark:border-slate-700 space-y-3">
   300→          <div className="flex flex-col sm:flex-row gap-3">
   301→            <div className="relative flex-1">
   302→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   303→              <input
   304→                type="text"
   305→                placeholder="Search by name, staff ID, email, or zone..."
   306→                value={search}
   307→                onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   308→                className="w-full rounded-lg border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900 pl-10 pr-4 py-2 text-sm text-slate-900 dark:text-white placeholder:text-slate-400 dark:placeholder:text-slate-500 focus:outline-none focus:ring-2 focus:ring-emerald-500 focus:border-transparent transition-shadow"
   309→              />
   310→            </div>
   311→            <button
   312→              onClick={() => setShowFilters((v) => !v)}
   313→              className="inline-flex items-center gap-2 rounded-lg border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900 px-3 py-2 text-sm font-medium text-slate-700 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors"
   314→            >
   315→              <Filter className="w-4 h-4" />
   316→              Filters
   317→            </button>
   318→          </div>
   319→
   320→          {showFilters && (
   321→            <div className="flex flex-wrap gap-3">
   322→              <select
   323→                value={roleFilter}
   324→                onChange={(e) => { setRoleFilter(e.target.value as 'All' | UserRole); setPage(1); }}
   325→                className="rounded-lg border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900 px-3 py-2 text-sm text-slate-700 dark:text-slate-300 focus:outline-none focus:ring-2 focus:ring-emerald-500"
   326→              >
   327→                <option value="All">All Roles</option>
   328→                <option value="Administrator">Administrator</option>
   329→                <option value="Revenue Officer">Revenue Officer</option>
   330→                <option value="Field Collector">Field Collector</option>
   331→                <option value="Auditor">Auditor</option>
   332→                <option value="Finance Manager">Finance Manager</option>
   333→              </select>
   334→              <select
   335→                value={statusFilter}
   336→                onChange={(e) => { setStatusFilter(e.target.value as 'All' | UserStatus); setPage(1); }}
   337→                className="rounded-lg border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900 px-3 py-2 text-sm text-slate-700 dark:text-slate-300 focus:outline-none focus:ring-2 focus:ring-emerald-500"
   338→              >
   339→                <option value="All">All Statuses</option>
   340→                <option value="Active">Active</option>
   341→                <option value="Inactive">Inactive</option>
   342→                <option value="Suspended">Suspended</option>
   343→              </select>
   344→            </div>
   345→          )}
   346→        </div>
   347→
   348→        {/* Table */}
   349→        <div className="overflow-x-auto max-h-96 overflow-y-auto">
   350→          <table className="w-full text-left">
   351→            <thead className="bg-slate-50 dark:bg-slate-900/50 sticky top-0 z-10">
   352→              <tr className="border-b border-slate-200 dark:border-slate-700">
   353→                <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Staff ID</th>
   354→                <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Name</th>
   355→                <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Role</th>
   356→                <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Zone / Ward</th>
   357→                <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Status</th>
   358→                <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Last Login</th>
   359→                <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3 text-center">Actions</th>
   360→              </tr>
   361→            </thead>
   362→            <tbody className="divide-y divide-slate-100 dark:divide-slate-700/60">
   363→              {paginated.length === 0 ? (
   364→                <tr>
   365→                  <td colSpan={7} className="text-center py-12 text-sm text-slate-400 dark:text-slate-500">
   366→                    No users found matching your criteria.
   367→                  </td>
   368→                </tr>
   369→              ) : (
   370→                paginated.map((u) => {
   371→                  const StatusIcon = statusConfig[u.status].icon;
   372→                  return (
   373→                    <tr key={u.id} className="hover:bg-slate-50 dark:hover:bg-slate-700/30 transition-colors">
   374→                      <td className="px-4 py-3 text-sm font-mono text-slate-900 dark:text-white">{u.staffId}</td>
   375→                      <td className="px-4 py-3">
   376→                        <div className="flex items-center gap-2.5">
   377→                          <div className="w-8 h-8 rounded-full bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center text-emerald-700 dark:text-emerald-400 text-xs font-bold shrink-0">
   378→                            {u.firstName[0]}{u.lastName[0]}
   379→                          </div>
   380→                          <div>
   381→                            <p className="text-sm font-medium text-slate-900 dark:text-white">{u.firstName} {u.lastName}</p>
   382→                            <p className="text-xs text-slate-500 dark:text-slate-400">{u.email}</p>
   383→                          </div>
   384→                        </div>
   385→                      </td>
   386→                      <td className="px-4 py-3">
   387→                        <span className={`inline-flex text-xs font-medium px-2 py-0.5 rounded-full ${roleIcon[u.role]}`}>
   388→                          {u.role}
   389→                        </span>
   390→                      </td>
   391→                      <td className="px-4 py-3">
   392→                        <p className="text-sm text-slate-700 dark:text-slate-300">{u.zone}</p>
   393→                        <p className="text-xs text-slate-400 dark:text-slate-500">{u.ward}</p>
   394→                      </td>
   395→                      <td className="px-4 py-3">
   396→                        <span className={`inline-flex items-center gap-1 text-xs font-medium px-2 py-0.5 rounded-full ${statusConfig[u.status].pill}`}>
   397→                          <StatusIcon className="w-3 h-3" />
   398→                          {u.status}
   399→                        </span>
   400→                      </td>
   401→                      <td className="px-4 py-3">
   402→                        <div className="text-sm text-slate-600 dark:text-slate-400">{u.lastLogin.split(' ')[0]}</div>
   403→                        <div className="text-xs text-slate-400 dark:text-slate-500">{u.lastLogin.split(' ')[1]}</div>
   404→                      </td>
   405→                      <td className="px-4 py-3">
   406→                        <div className="flex items-center justify-center gap-1">
   407→                          <button
   408→                            onClick={() => setViewUser(u)}
   409→                            className="p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   410→                            title="View user"
   411→                          >
   412→                            <Eye className="w-4 h-4" />
   413→                          </button>
   414→                          <button
   415→                            onClick={() => openEdit(u)}
   416→                            className="p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   417→                            title="Edit user"
   418→                          >
   419→                            <Pencil className="w-4 h-4" />
   420→                          </button>
   421→                          <button
   422→                            onClick={() => toggleStatus(u.id)}

   424→                            title={u.status === 'Active' ? 'Deactivate' : 'Activate'}
   425→                          >
   426→                            <Shield className="w-4 h-4" />

   431→                            title="Delete user"
   432→                          >
   433→                            <Trash2 className="w-4 h-4" />
   434→                          </button>
   435→                        </div>
   436→                      </td>
   437→                    </tr>
   438→                  );

   441→            </tbody>
   442→          </table>
   443→        </div>
   444→
   445→        {/* Pagination */}
   446→        <div className="flex items-center justify-between px-4 py-3 border-t border-slate-200 dark:border-slate-700">
   447→          <p className="text-sm text-slate-500 dark:text-slate-400">
