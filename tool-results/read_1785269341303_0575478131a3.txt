   181→// ─── Helpers ─────────────────────────────────────────────────────────────────
   182→
   183→function formatCurrency(amount: number): string {
   184→  return `GH₵ ${amount.toLocaleString()}`;
   185→}

   187→// ─── Component ───────────────────────────────────────────────────────────────
   188→
   189→export function RateConfigPage() {
   190→  const [rates, setRates] = useState<RateType[]>(initialRates);
   191→  const [expandedRows, setExpandedRows] = useState<Set<string>>(new Set());
   192→  const [searchQuery, setSearchQuery] = useState('');
   193→  const [categoryFilter, setCategoryFilter] = useState<string>('All');
   194→  const [statusFilter, setStatusFilter] = useState<string>('All');
   195→  const [showModal, setShowModal] = useState(false);
   196→  const [editingRate, setEditingRate] = useState<RateType | null>(null);
   197→  const [formData, setFormData] = useState<RateFormData>({
   198→    id: '',
   199→    name: '',
   200→    category: 'Business',
   201→    subRateName: '',
   202→    amount: 0,
   203→    effectiveDate: '',
   204→    expiryDate: '',
   205→    status: 'Active',
   206→    annualIncrement: 0,
   207→  });
   208→
   209→  // ── Filtering ───────────────────────────────────────────────────────────
   210→
   211→  const filtered = useMemo(() => {
   212→    return rates.filter((r) => {
   213→      const matchSearch =
   214→        searchQuery === '' ||
   215→        r.name.toLowerCase().includes(searchQuery.toLowerCase());
   216→      const matchCategory =
   217→        categoryFilter === 'All' || r.category === categoryFilter;
   218→      const matchStatus =
   219→        statusFilter === 'All' || r.status === statusFilter;
   220→      return matchSearch && matchCategory && matchStatus;
   221→    });
   222→  }, [rates, searchQuery, categoryFilter, statusFilter]);
   223→
   224→  // ── Toggle expand ───────────────────────────────────────────────────────
   225→
   226→  const toggleExpand = (id: string) => {
   227→    setExpandedRows((prev) => {
   228→      const next = new Set(prev);
   229→      if (next.has(id)) {
   230→        next.delete(id);
   231→      } else {
   232→        next.add(id);
   233→      }
   234→      return next;
   235→    });
   236→  };
   237→
   238→  // ── Toggle status ──────────────────────────────────────────────────────
   239→
   240→  const toggleStatus = (id: string) => {
   241→    setRates((prev) =>
   242→      prev.map((r) =>
   243→        r.id === id
   244→          ? { ...r, status: r.status === 'Active' ? 'Inactive' : 'Active' }
   245→          : r
   246→      )
   247→    );
   248→  };
   249→
   250→  // ── Delete ──────────────────────────────────────────────────────────────
   251→
   252→  const handleDelete = (id: string) => {
   253→    setRates((prev) => prev.filter((r) => r.id !== id));
   254→  };
   255→
   256→  // ── Modal open for Add ──────────────────────────────────────────────────
   257→
   258→  const openAddModal = () => {
   259→    setEditingRate(null);
   260→    setFormData({
   261→      id: '',
   262→      name: '',
   263→      category: 'Business',
   264→      subRateName: '',
   265→      amount: 0,
   266→      effectiveDate: '',
   267→      expiryDate: '',
   268→      status: 'Active',
   269→      annualIncrement: 0,
   270→    });
   271→    setShowModal(true);
   272→  };
   273→
   274→  // ── Modal open for Edit ─────────────────────────────────────────────────
   275→
   276→  const openEditModal = (rate: RateType) => {
   277→    setEditingRate(rate);
   278→    setFormData({
   279→      id: rate.id,
   280→      name: rate.name,
   281→      category: rate.category,
   282→      subRateName: rate.subRates[0]?.name ?? '',
   283→      amount: rate.subRates[0]?.amount ?? 0,
   284→      effectiveDate: rate.effectiveDate,
   285→      expiryDate: rate.expiryDate,
   286→      status: rate.status,
   287→      annualIncrement: rate.annualIncrement,
   288→    });
   289→    setShowModal(true);
   290→  };
   291→
   292→  // ── Save ────────────────────────────────────────────────────────────────
   293→
   294→  const handleSave = () => {
   295→    if (!formData.name.trim() || !formData.subRateName.trim() || formData.amount <= 0) return;
   296→
   297→    if (editingRate) {
   298→      // Update existing rate
   299→      setRates((prev) =>
   300→        prev.map((r) =>
   301→          r.id === editingRate.id
   302→            ? {
   303→                ...r,
   304→                name: formData.name,
   305→                category: formData.category,
   306→                subRates: [{ id: r.subRates[0]?.id ?? `${r.id}-sub`, name: formData.subRateName, amount: formData.amount }],
   307→                effectiveDate: formData.effectiveDate,
   308→                expiryDate: formData.expiryDate,
   309→                status: formData.status,
   310→                annualIncrement: formData.annualIncrement,
   311→              }
   312→            : r
   313→        )
   314→      );
   315→    } else {
   316→      // Add new rate
   317→      const newId = `RT-${String(rates.length + 1).padStart(3, '0')}`;
   318→      const newRate: RateType = {
   319→        id: newId,
   320→        name: formData.name,
   321→        category: formData.category,
   322→        subRates: [{ id: `${newId}-sub`, name: formData.subRateName, amount: formData.amount }],
   323→        effectiveDate: formData.effectiveDate,
   324→        expiryDate: formData.expiryDate,
   325→        status: formData.status,
   326→        annualIncrement: formData.annualIncrement,
   327→      };
   328→      setRates((prev) => [...prev, newRate]);
   329→    }
   330→
   331→    setShowModal(false);
   332→  };
   333→
   334→  // ── CSS classes ──────────────────────────────────────────────────────────
   335→
   336→  const inputClass =
   337→    'w-full rounded-lg border border-slate-300 dark:border-slate-600 bg-white dark:bg-slate-800 px-4 py-2.5 text-sm focus:ring-2 focus:ring-emerald-500 focus:border-emerald-500 outline-none transition';
   338→  const labelClass =
   339→    'block text-sm font-medium text-slate-700 dark:text-slate-300 mb-1.5';
   340→  const btnPrimary =
   341→    'inline-flex items-center gap-2 bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-medium px-4 py-2.5 rounded-lg transition-colors whitespace-nowrap';
   342→  const btnSecondary =
   343→    'inline-flex items-center gap-2 bg-slate-100 dark:bg-slate-700 hover:bg-slate-200 dark:hover:bg-slate-600 text-slate-700 dark:text-slate-200 text-sm font-medium px-4 py-2.5 rounded-lg transition-colors whitespace-nowrap';
   344→
   345→  // ── Compute display amount from sub-rates ────────────────────────────────
   346→
   347→  const getDisplayAmount = (rate: RateType) => {
   348→    if (rate.subRates.length === 1) return formatCurrency(rate.subRates[0].amount);
   349→    const min = Math.min(...rate.subRates.map((s) => s.amount));
   350→    const max = Math.max(...rate.subRates.map((s) => s.amount));
   351→    return `${formatCurrency(min)} – ${formatCurrency(max)}`;
   352→  };
   353→
   354→  // ══════════════════════════════════════════════════════════════════════════
   355→  //  RENDER
   356→  // ══════════════════════════════════════════════════════════════════════════
   357→
   358→  return (
   359→    <div className="space-y-6">
   360→      {/* ── Header ──────────────────────────────────────────────────────── */}
   361→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   362→        <div>
   363→          <h1 className="text-2xl font-bold text-slate-900 dark:text-white">
   364→            Rate Configuration
   365→          </h1>
   366→          <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">
   367→            Manage revenue rate types, sub-rates, and fee structures.
   368→          </p>
   369→        </div>
   370→        <button onClick={openAddModal} className={btnPrimary}>
   371→          <Plus className="w-4 h-4" />
   372→          Add New Rate Type
   373→        </button>
   374→      </div>
   375→
   376→      {/* ── Filters ────────────────────────────────────────────────────── */}
   377→      <div className="flex flex-col sm:flex-row gap-3">
   378→        <div className="relative flex-1">
   379→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   380→          <input
   381→            type="text"
   382→            placeholder="Search rate types..."
   383→            value={searchQuery}
   384→            onChange={(e) => setSearchQuery(e.target.value)}
   385→            className={`${inputClass} pl-10`}
   386→          />
   387→        </div>
   388→        <select
   389→          value={categoryFilter}
   390→          onChange={(e) => setCategoryFilter(e.target.value)}
   391→          className={`${inputClass} w-full sm:w-44`}
   392→        >
   393→          <option value="All">All Categories</option>
   394→          <option value="Business">Business</option>
   395→          <option value="Property">Property</option>
   396→        </select>
   397→        <select
   398→          value={statusFilter}
   399→          onChange={(e) => setStatusFilter(e.target.value)}
   400→          className={`${inputClass} w-full sm:w-44`}
   401→        >
   402→          <option value="All">All Statuses</option>
   403→          <option value="Active">Active</option>
   404→          <option value="Inactive">Inactive</option>
   405→        </select>
   406→      </div>
   407→
   408→      {/* ── Table ───────────────────────────────────────────────────────── */}
   409→      <div className="rounded-xl border border-slate-200 dark:border-slate-700 overflow-hidden">
   410→        <div className="overflow-x-auto">
   411→          <table className="w-full text-sm">
   412→            <thead>
   413→              <tr className="bg-slate-50 dark:bg-slate-800/60 border-b border-slate-200 dark:border-slate-700">
   414→                <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap w-10" />
   415→                <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   416→                  Rate Name
   417→                </th>
   418→                <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   419→                  Category
   420→                </th>
   421→                <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   422→                  Amount
   423→                </th>
   424→                <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   425→                  Status
   426→                </th>
   427→                <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap hidden md:table-cell">
   428→                  Effective Date
   429→                </th>
   430→                <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap hidden md:table-cell">
   431→                  Expiry Date
   432→                </th>
   433→                <th className="text-right px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   434→                  Actions
   435→                </th>
   436→              </tr>
   437→            </thead>
   438→            <tbody className="divide-y divide-slate-100 dark:divide-slate-700">
   439→              {filtered.length === 0 ? (
   440→                <tr>
   441→                  <td
   442→                    colSpan={8}
   443→                    className="text-center py-12 text-slate-400 dark:text-slate-500"
   444→                  >
   445→                    <Settings2 className="w-10 h-10 mx-auto mb-3 opacity-40" />
   446→                    No rate types found matching your criteria.
   447→                  </td>
   448→                </tr>
   449→              ) : (
   450→                filtered.map((rate) => {
   451→                  const isExpanded = expandedRows.has(rate.id);
   452→                  return (
   453→                    <>
   454→                      {/* Parent row */}
   455→                      <tr
   456→                        key={rate.id}
   457→                        className="hover:bg-slate-50 dark:hover:bg-slate-800/40 transition-colors cursor-pointer"
   458→                      >
   459→                        <td
   460→                          className="px-4 py-3 text-slate-400"
   461→                          onClick={() => toggleExpand(rate.id)}
   462→                        >
   463→                          {rate.subRates.length > 1 ? (
   464→                            isExpanded ? (
   465→                              <ChevronDown className="w-4 h-4" />
   466→                            ) : (
   467→                              <ChevronRight className="w-4 h-4" />
   468→                            )
   469→                          ) : (
   470→                            <span className="w-4 h-4 inline-block" />
   471→                          )}
   472→                        </td>
   473→                        <td
   474→                          className="px-4 py-3 font-medium text-slate-900 dark:text-white whitespace-nowrap"
   475→                          onClick={() => toggleExpand(rate.id)}
   476→                        >
   477→                          {rate.name}
   478→                        </td>
   479→                        <td className="px-4 py-3 whitespace-nowrap">
   480→                          <span
   481→                            className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-semibold ${
   482→                              rate.category === 'Business'
   483→                                ? 'bg-blue-100 text-blue-700 dark:bg-blue-900/40 dark:text-blue-400'
   484→                                : 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400'
   485→                            }`}
   486→                          >
   487→                            {rate.category}
   488→                          </span>
   489→                        </td>
   490→                        <td className="px-4 py-3 font-medium text-slate-900 dark:text-white whitespace-nowrap">
   491→                          {getDisplayAmount(rate)}
   492→                        </td>
   493→                        <td className="px-4 py-3 whitespace-nowrap">
   494→                          <span
   495→                            className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-semibold ${
   496→                              rate.status === 'Active'
   497→                                ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400'
   498→                                : 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400'
   499→                            }`}
   500→                          >
   501→                            {rate.status}
   502→                          </span>
   503→                        </td>
   504→                        <td className="px-4 py-3 text-slate-500 dark:text-slate-400 whitespace-nowrap hidden md:table-cell">
   505→                          {rate.effectiveDate}
   506→                        </td>
   507→                        <td className="px-4 py-3 text-slate-500 dark:text-slate-400 whitespace-nowrap hidden md:table-cell">
   508→                          {rate.expiryDate}
   509→                        </td>
   510→                        <td className="px-4 py-3 text-right whitespace-nowrap">
   511→                          <div className="inline-flex items-center gap-1">
   512→                            <button
   513→                              onClick={() => openEditModal(rate)}
   514→                              className="p-1.5 rounded-md text-slate-400 hover:text-emerald-600 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   515→                              title="Edit"
   516→                            >
   517→                              <Pencil className="w-4 h-4" />
   518→                            </button>
   519→                            <button
   520→                              onClick={() => toggleStatus(rate.id)}
   521→                              className="p-1.5 rounded-md text-slate-400 hover:text-amber-600 hover:bg-amber-50 dark:hover:bg-amber-900/20 transition-colors"
   522→                              title="Toggle Status"
   523→                            >
   524→                              <Power className="w-4 h-4" />
   525→                            </button>
   526→                            <button
   527→                              onClick={() => handleDelete(rate.id)}
   528→                              className="p-1.5 rounded-md text-slate-400 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-900/20 transition-colors"
   529→                              title="Delete"
   530→                            >
   531→                              <Trash2 className="w-4 h-4" />
   532→                            </button>
   533→                          </div>
   534→                        </td>
   535→                      </tr>
   536→
   537→                      {/* Expanded sub-rate rows */}
   538→                      {isExpanded && rate.subRates.length > 1 && (
   539→                        <tr key={`${rate.id}-sub`} className="bg-slate-50/60 dark:bg-slate-800/30">
   540→                          <td />
