   118→   118→
   119→   119→function loadUsers(): User[] {
   120→   120→  if (typeof window === 'undefined') return defaultUsers;
   121→   121→  try {
   122→   122→    const stored = localStorage.getItem(USERS_STORAGE_KEY);
   123→   123→    if (stored) {
   124→   124→      const parsed = JSON.parse(stored);
   125→   125→      if (Array.isArray(parsed) && parsed.length > 0) {
   126→   126→        // Migrate: ensure every user has required fields
   127→   127→        return parsed.map((u: Partial<User>) => ({
   128→   128→          ...defaultUsers[0],
   129→   129→          ...u,
   130→   130→          lastLogin: u.lastLogin || 'Never',
   131→   131→          dateCreated: u.dateCreated || new Date().toISOString().split('T')[0],
   132→   132→          accessiblePages: u.accessiblePages || ROLE_DEFAULT_PAGES[u.role as UserRole] || [],
   133→   133→        })) as User[];
   134→   134→      }
   135→   135→    }
   136→   136→  } catch { /* ignore */ }
   137→   137→  return defaultUsers;
   138→   138→}

   140→   140→function saveUsers(users: User[]) {
   141→   141→  if (typeof window === 'undefined') return;
   142→   142→  try { localStorage.setItem(USERS_STORAGE_KEY, JSON.stringify(users)); } catch { /* ignore */ }
   143→   143→}
