   580→                  </button>

   584→                  </button>
   585→                </div>
   586→              </div>

   588→              {/* Info Grid */}

   602→              </div>
   603→
   604→              {/* Line Items */}
   605→              <div>
   606→                <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium mb-3">Line Items</p>
   607→                <div className="rounded-lg border border-slate-200 dark:border-slate-700 overflow-hidden">
   608→                  <table className="w-full text-left text-sm">
   609→                    <thead className="bg-slate-50 dark:bg-slate-900/50">
   610→                      <tr className="border-b border-slate-200 dark:border-slate-700">

   626→                    </tbody>
   627→                    <tfoot className="bg-slate-50 dark:bg-slate-900/50">
   628→                      <tr className="border-t border-slate-200 dark:border-slate-700">
   629→                        <td colSpan={3} className="px-3 py-2.5 text-sm font-semibold text-slate-700 dark:text-slate-300 text-right">Total Paid</td>
   630→                        <td className="px-3 py-2.5 text-sm font-bold text-emerald-600 dark:text-emerald-400 text-right">{fmtCurrency(selectedReceipt.totalPaid)}</td>

   633→                  </table>
   634→                </div>
   635→              </div>

   638→              <div className="flex items-center gap-6 text-sm text-slate-500 dark:text-slate-400">

   641→              </div>
   642→            </div>
   643→          </div>
   644→        </div>
   645→      )}
   646→    </div>
   647→  );
   648→}
