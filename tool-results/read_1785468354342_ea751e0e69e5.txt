   784→            </div>
   785→          </div>
   786→        </div>
   787→      )}

   802→                <X className="w-5 h-5" />

   806→              <div className="grid grid-cols-2 gap-4">
   807→                <ViewField label="Username" value={viewUser.username} />

   813→                <ViewField label="Ward" value={viewUser.ward} />
   814→              </div>
   815→              <div className="border-t border-slate-200 dark:border-slate-700 pt-4 space-y-3">
   816→                <div className="flex items-center gap-2 text-sm text-slate-600 dark:text-slate-400">
   817→                  <Mail className="w-4 h-4" />{viewUser.email}
   818→                </div>

   820→                  <Phone className="w-4 h-4" />{viewUser.phone}
   821→                </div>
   822→                <div className="flex items-center gap-2 text-sm text-slate-600 dark:text-slate-400">
   823→                  <Clock className="w-4 h-4" />Last login: {viewUser.lastLogin}
   824→                </div>
   825→                <div className="flex items-center gap-2 text-sm text-slate-600 dark:text-slate-400">
   826→                  <MapPin className="w-4 h-4" />Created: {viewUser.dateCreated}
   827→                </div>
   828→              </div>
   829→              {/* Accessible Pages */}
   830→              <div className="border-t border-slate-200 dark:border-slate-700 pt-4">
   831→                <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium mb-2">Accessible Navigation Pages</p>
   832→                <div className="flex flex-wrap gap-1.5">
   833→                  {ALL_RMS_PAGES.map((p) => {
   834→                    const hasAccess = viewUser.accessiblePages?.includes(p.page);
   835→                    return (
   836→                      <span
   837→                        key={p.page}
   838→                        className={`inline-flex items-center px-2 py-0.5 rounded text-[10px] font-medium ${

   841→                            : 'bg-slate-100 text-slate-400 dark:bg-slate-700 dark:text-slate-500 line-through'
   842→                        }`}
   843→                      >
   844→                        {p.label}
   845→                      </span>
   846→                    );
   847→                  })}
   848→                </div>
   849→              </div>
   850→
   851→              {/* Login As Button */}
   852→              {viewUser.status !== 'Suspended' && (
   853→                <div className="border-t border-slate-200 dark:border-slate-700 pt-4">
   854→                  <button

   857→                  >

   859→                    Login as {viewUser.firstName} {viewUser.lastName}

   867→          </div>

   869→      )}
   870→    </div>
   871→  );
   872→}
