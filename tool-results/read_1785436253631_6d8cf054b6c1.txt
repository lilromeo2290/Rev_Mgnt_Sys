    92→function formatCurrency(amount: number): string {

   100→  const [searchQuery, setSearchQuery] = useState('');

   103→  const [revenueAreaFilter, setRevenueAreaFilter] = useState<string>('All');

   105→  const [dateTo, setDateTo] = useState('');

   107→  const [showModal, setShowModal] = useState(false);
   108→  const [showBulkModal, setShowBulkModal] = useState(false);
   109→  const [bulkForm, setBulkForm] = useState({
   110→    entityType: 'All' as 'All' | 'Business' | 'Property',
   111→    category: 'All',
   112→    revenueItem: '',
   113→    dueDate: '',
   114→  });

   122→  const [formData, setFormData] = useState<BillFormData>({
   123→    entityName: '',
   124→    entityType: 'Business',
   125→    revenueItem: '',
   126→    amount: 0,
   127→    previousBalance: 0,
   128→    penalty: 0,
   129→    dueDate: '',
   130→  });

   132→  // ── Filtering ───────────────────────────────────────────────────────────
   133→
   134→  const filtered = useMemo(() => {
   135→    return bills.filter((b) => {
   136→      const matchSearch =
   137→        searchQuery === '' ||
   138→        b.entityName.toLowerCase().includes(searchQuery.toLowerCase()) ||
   139→        b.billNumber.toLowerCase().includes(searchQuery.toLowerCase()) ||
   140→        b.revenueItem.toLowerCase().includes(searchQuery.toLowerCase());
   141→      const matchStatus =
   142→        statusFilter === 'All' || b.status === statusFilter;
   143→      const matchCategory =
   144→        categoryFilter === 'All' || b.category === categoryFilter;
   145→      const matchRevenueArea =
   146→        revenueAreaFilter === 'All' ||
   147→        (b.entityType === revenueAreaFilter);
   148→      const matchDateFrom =
   149→        dateFrom === '' || b.date >= dateFrom;
   150→      const matchDateTo =
   151→        dateTo === '' || b.date <= dateTo;
   152→      return matchSearch && matchStatus && matchCategory && matchRevenueArea && matchDateFrom && matchDateTo;
   153→    });
   154→  }, [bills, searchQuery, statusFilter, categoryFilter, revenueAreaFilter, dateFrom, dateTo]);
   155→
   156→  // ── Pagination ──────────────────────────────────────────────────────────
   157→
   158→  const totalPages = Math.max(1, Math.ceil(filtered.length / itemsPerPage));
   159→  const safeCurrentPage = Math.min(currentPage, totalPages);
   160→  const startIdx = (safeCurrentPage - 1) * itemsPerPage;
   161→  const paged = filtered.slice(startIdx, startIdx + itemsPerPage);
   162→  const showingFrom = filtered.length === 0 ? 0 : startIdx + 1;
   163→  const showingTo = Math.min(startIdx + itemsPerPage, filtered.length);

   167→  const StatusBadge = ({ status }: { status: Bill['status'] }) => {
   168→    const styles: Record<Bill['status'], string> = {
   169→      Paid: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400',
   170→      Partial: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400',
   171→      Unpaid: 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400',
   172→      Overdue: 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400',
   173→    };
   174→    return (
   175→      <span className={`inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-xs font-semibold ${styles[status]}`}>
   176→        {status === 'Overdue' && (
   177→          <span className="relative flex h-2 w-2">
   178→            <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-red-400 opacity-75" />
   179→            <span className="relative inline-flex rounded-full h-2 w-2 bg-red-500" />
   180→          </span>
   181→        )}
   182→        {status}
   183→      </span>
   184→    );
   185→  };
   186→
   187→  // ── Stats ──────────────────────────────────────────────────────────────
   188→
   189→  const stats = useMemo(() => {
   190→    const totalBilled = bills.reduce((sum, b) => sum + b.totalDue, 0);
   191→    const paid = bills
   192→      .filter((b) => b.status === 'Paid')
   193→      .reduce((sum, b) => sum + b.totalDue, 0);
   194→    const outstanding = totalBilled - paid;
   195→    const overdue = bills
   196→      .filter((b) => b.status === 'Overdue')
   197→      .reduce((sum, b) => sum + b.totalDue, 0);
   198→    return {
   199→      total: bills.length,
   200→      totalBilled,
   201→      outstanding,
   202→      overdue,
   203→    };
   204→  }, [bills]);

   206→  // ── Auto-fill amount when revenue item changes ──────────────────────────
   207→
   208→  const handleRevenueItemChange = (item: string) => {
   209→    const amount = RATE_AMOUNTS[item] || 0;
   210→    const penalty = Math.round(amount * 0.05 * 100) / 100;
   211→    setFormData((p) => ({
   212→      ...p,
   213→      revenueItem: item,
   214→      amount,
   215→      penalty,
   216→    }));

   223→
   224→    const newBillNumber = `BILL-2024-${String(bills.length + 156).padStart(4, '0')}`;

   235→      amount: formData.amount,
   236→      previousBalance: formData.previousBalance,
   237→      penalty: formData.penalty,

   239→      status: 'Unpaid',
   240→      dueDate: formData.dueDate || '',

   271→        (b) => b.entityName === e.name && b.revenueItem === bulkForm.revenueItem,

   273→      return !exists;
   274→    }).length;

   276→
   277→  // ── Bulk generate bills ────────────────────────────────────────────────
   278→
   279→  const handleBulkGenerate = () => {

   281→
   282→    setBulkProgress('generating');
   283→
   284→    const amount = RATE_AMOUNTS[bulkForm.revenueItem] || 0;
   285→    const penalty = Math.round(amount * 0.05 * 100) / 100;
   286→
   287→    setTimeout(() => {
   288→      const eligibleEntities = ENTITIES.filter((e) => {
   289→        if (bulkForm.entityType !== 'All' && e.type !== bulkForm.entityType) return false;
   290→        if (bulkForm.category !== 'All' && e.category !== bulkForm.category) return false;
   291→        const exists = bills.find(
   292→          (b) => b.entityName === e.name && b.revenueItem === bulkForm.revenueItem,
   293→        );
   294→        return !exists;
   295→      });
   296→
   297→      const newBills: Bill[] = eligibleEntities.map((entity, idx) => {
   298→        const totalDue = amount + penalty;
   299→        return {
   300→          id: String(Date.now() + idx),
   301→          billNumber: `BILL-2024-${String(bills.length + 156 + idx).padStart(4, '0')}`,
   302→          date: new Date().toISOString().split('T')[0],
   303→          entityName: entity.name,
   304→          entityType: entity.type as 'Business' | 'Property',
   305→          category: entity.category || 'General',
   306→          revenueItem: bulkForm.revenueItem,
   307→          amount,
   308→          previousBalance: 0,
   309→          penalty,
   310→          totalDue,
   311→          status: 'Unpaid' as const,
   312→          dueDate: bulkForm.dueDate || new Date(Date.now() + 30 * 86400000).toISOString().split('T')[0],
   313→        };
   314→      });
   315→
   316→      setBills((prev) => [...newBills.reverse(), ...prev]);
   317→      setBulkGeneratedCount(newBills.length);
   318→      setBulkProgress('done');
   319→      setCurrentPage(1);
   320→    }, 800);

   356→        fontSize: 10,

   360→      return '';

   364→  const getBillBarcodeData = (bill: Bill): string => {
   365→    return encodeBarcodeData({
   366→      type: 'INVOICE',
   367→      refNo: bill.billNumber,

   369→      entityType: bill.entityType,
   370→      amount: bill.totalDue,
   371→      date: bill.date,
   372→      revenueItem: bill.revenueItem,
   373→      status: bill.status,
