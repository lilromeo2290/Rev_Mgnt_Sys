    72→

    75→const defaultForm = {
    76→  regNumber: '',
    77→  name: '',
    78→  ownerName: '',
    79→  type: '',
    80→  category: '',
    81→  subCategory: '',

    98→  houseNo: '',

   100→  localityCode: '',

   106→  excludedFromFees: false,
   107→  ownerAddress: '',
   108→  ownerGps: '',

   110→  comments: '',
   111→};
   112→
   113→export function BusinessesPage() {
   114→  const [view, setView] = useState<'list' | 'form'>('list');
   115→  const [editingRegNumber, setEditingRegNumber] = useState<string | null>(null);
   116→  const [searchQuery, setSearchQuery] = useState('');
   117→  const [statusFilter, setStatusFilter] = useState<string>('All');
   118→  const [typeFilter, setTypeFilter] = useState<string>('All');
   119→  const [currentPage, setCurrentPage] = useState(1);
   120→  const [locatingBusiness, setLocatingBusiness] = useState(false);
   121→  const [locatingOwner, setLocatingOwner] = useState(false);
   122→  const [businesses, setBusinesses] = useLocalStorage<Business[]>('rms-businesses', mockBusinesses);
   123→  const itemsPerPage = 10;
   124→
   125→  // ── Form State ───────────────────────────────────────────────────────────
   126→  const [form, setForm] = useState({ ...defaultForm });
   127→
   128→  // ── Geolocation: Fetch GPS from device location ─────────────────────
   129→  const fetchGpsFromLocation = (
   130→    target: 'business' | 'owner'
   131→  ) => {
   132→    const setLoading = target === 'business' ? setLocatingBusiness : setLocatingOwner;
   133→
   134→    if (!navigator.geolocation) {
   135→      alert('Geolocation is not supported by your browser. Please enter the GPS address manually.');
   136→      return;
   137→    }
   138→
   139→    setLoading(true);
   140→
   141→    navigator.geolocation.getCurrentPosition(
   142→      (position) => {
   143→        const { latitude, longitude } = position.coords;
   144→        const gpsString = `${latitude.toFixed(6)}, ${longitude.toFixed(6)}`;
   145→        if (target === 'business') {
   146→          setForm((prev) => ({ ...prev, gpsAddress: gpsString }));
   147→        } else {
   148→          setForm((prev) => ({ ...prev, ownerGps: gpsString }));
   149→        }
   150→        setLoading(false);
   151→      },
   152→      (error) => {
   153→        let message = 'Unable to retrieve location.';
   154→        switch (error.code) {
   155→          case error.PERMISSION_DENIED:
   156→            message = 'Location permission denied. Please allow location access in your browser settings.';
   157→            break;
   158→          case error.POSITION_UNAVAILABLE:
   159→            message = 'Location information is unavailable. Please try again or enter manually.';
   160→            break;
   161→          case error.TIMEOUT:
   162→            message = 'Location request timed out. Please try again.';
   163→            break;
   164→        }
   165→        alert(message);
   166→        setLoading(false);
   167→      },
   168→      { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
   169→    );
   170→  };
   171→
   172→  // ── Derived categories based on selected business type ───────────────────
   173→  const availableCategories: FeeCategory[] = form.type
   174→    ? (BUSINESS_CLASS_CATEGORIES[form.type] || [])
   175→    : [];
   176→
   177→  // ── Get fee details for selected category ──────────────────────────────
   178→  const selectedCategoryFee = availableCategories.find(
   179→    (c) => c.name === form.category
   180→  );
   181→
   182→  // ── Filtering & Pagination ───────────────────────────────────────────────
   183→  const filtered = businesses.filter((b) => {
   184→    const matchSearch =
   185→      searchQuery === '' ||
   186→      b.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   187→      b.owner.toLowerCase().includes(searchQuery.toLowerCase()) ||
   188→      b.regNumber.toLowerCase().includes(searchQuery.toLowerCase());
   189→    const matchStatus = statusFilter === 'All' || b.status === statusFilter;
   190→    const matchType = typeFilter === 'All' || b.type === typeFilter;
   191→    return matchSearch && matchStatus && matchType;
   192→  });
   193→
   194→  const totalPages = Math.max(1, Math.ceil(filtered.length / itemsPerPage));
   195→  const safeCurrentPage = Math.min(currentPage, totalPages);
   196→  const startIdx = (safeCurrentPage - 1) * itemsPerPage;
   197→  const paged = filtered.slice(startIdx, startIdx + itemsPerPage);
   198→  const showingFrom = filtered.length === 0 ? 0 : startIdx + 1;
   199→  const showingTo = Math.min(startIdx + itemsPerPage, filtered.length);
   200→
   201→  // ── Handlers ─────────────────────────────────────────────────────────────
   202→  const handleFormChange = (
   203→    e: React.ChangeEvent<
   204→      HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement
   205→    >
   206→  ) => {
   207→    const { name, type } = e.target;
   208→    setForm((prev) => {
   209→      const updated = { ...prev, [name]: type === 'checkbox' ? (e.target as HTMLInputElement).checked : (e.target as HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement).value };
   210→      // Reset category and sub-category when type changes
   211→      if (name === 'type') {
   212→        updated.category = '';
   213→        updated.subCategory = '';
   214→      }
   215→      return updated;
   216→    });
   217→  };
   218→
   219→  const handleSave = () => {
   220→    if (!form.name || !form.type) return;
   221→    const regNum = form.regNumber || `BIZ-${String(businesses.length + 1).padStart(4, '0')}`;
   222→    const newBusiness: Business = {
   223→      regNumber: regNum,
   224→      name: form.name,
   225→      owner: form.ownerName,
   226→      type: form.type,
   227→      category: form.category,
   228→      tin: form.tin,
   229→      status: (form.status as 'Active' | 'Inactive') || 'Active',
   230→      dateRegistered: form.dateRegistered || new Date().toISOString().split('T')[0],
   231→      ghanaCard: form.ghanaCard,
   232→      phone: form.phone,
   233→      email: form.email,
   234→      gpsAddress: form.gpsAddress,
   235→      digitalAddress: form.digitalAddress,
   236→      residentialAddress: form.residentialAddress,
   237→      businessAddress: form.businessAddress,
   238→      ward: form.ward,
   239→      electoralArea: form.electoralArea,
   240→      zone: form.zone,
   241→      revenueArea: form.revenueArea,
   242→      licenseNumber: form.licenseNumber,
   243→      subCategory: form.subCategory,
   244→      // New fields
   245→      streetName: form.streetName,
   246→      houseNo: form.houseNo,
   247→      streetCode: form.streetCode,
   248→      localityCode: form.localityCode,
   249→      daAssignmentNo: form.daAssignmentNo,
   250→      businessCertNo: form.businessCertNo,
   251→      businessPermit: form.businessPermit,
   252→      employees: form.employees,
   253→      yearEstablished: form.yearEstablished,
   254→      excludedFromFees: form.excludedFromFees,
   255→      ownerAddress: form.ownerAddress,
   256→      ownerGps: form.ownerGps,
   257→      ownerTin: form.ownerTin,
   258→      comments: form.comments,
   259→    };
   260→
   261→    if (editingRegNumber) {
   262→      setBusinesses((prev) =>
   263→        prev.map((b) =>
   264→          b.regNumber === editingRegNumber
   265→            ? { ...b, ...newBusiness, businessName: newBusiness.name, ownerName: newBusiness.owner, businessType: newBusiness.type, category: newBusiness.category, businessAddress: newBusiness.businessAddress }
   266→            : b
   267→        )
   268→      );
   269→    } else {
   270→      setBusinesses((prev) => [...prev, newBusiness]);
   271→    }
   272→
   273→    // Auto-generate business certificate
   274→    try {
   275→      const existingCerts = JSON.parse(localStorage.getItem('rms-business-certs') || '[]');
   276→      const certSeq = existingCerts.length + 1;
   277→      const today = new Date().toISOString().split('T')[0];
   278→      const expiryDate = new Date();
   279→      expiryDate.setFullYear(expiryDate.getFullYear() + 1);
   280→      const expiryStr = expiryDate.toISOString().split('T')[0];
   281→      const assemblyName = (() => {
   282→        try { const r = JSON.parse(localStorage.getItem('rms-settings-assembly') || '{}'); return r.name || 'Kumasi Metropolitan Assembly'; } catch { return 'Kumasi Metropolitan Assembly'; }
   283→      })();
   284→      const assemblyAddress = (() => {
   285→        try { const r = JSON.parse(localStorage.getItem('rms-settings-assembly') || '{}'); return r.address || ''; } catch { return ''; }
   286→      })();
   287→      const newCert = {
   288→        id: `CRT-${Date.now()}`,
   289→        certNumber: `CRT-${String(certSeq).padStart(4, '0')}`,
   290→        regNumber: regNum,
   291→        businessName: newBusiness.name,
   292→        ownerName: newBusiness.owner,
   293→        businessType: newBusiness.type,
   294→        category: newBusiness.category,
   295→        businessAddress: newBusiness.businessAddress,
   296→        dateRegistered: newBusiness.dateRegistered,
   297→        dateIssued: today,
   298→        expiryDate: expiryStr,
   299→        status: 'Active' as const,
   300→        assemblyName,
   301→        assemblyAddress,
   302→        tradingName: newBusiness.name,
   303→        receiptNumber: `RCT-${String(certSeq).padStart(4, '0')}`,
   304→      };
   305→      existingCerts.push(newCert);
   306→      localStorage.setItem('rms-business-certs', JSON.stringify(existingCerts));
   307→    } catch { /* cert generation failure should not block registration */ }
   308→
   309→    setEditingRegNumber(null);
   310→    setForm({ ...defaultForm });
   311→    setView('list');
   312→  };
   313→
   314→  const handleCancel = () => {
   315→    setEditingRegNumber(null);
   316→    setForm({ ...defaultForm });
   317→    setView('list');
   318→  };
   319→
   320→  const handleEdit = (biz: Business) => {
   321→    setEditingRegNumber(biz.regNumber);
   322→    setForm({
   323→      regNumber: biz.regNumber,
   324→      name: biz.name,
   325→      ownerName: biz.owner,
   326→      type: biz.type,
   327→      category: biz.category,
   328→      subCategory: biz.subCategory || '',
   329→      tin: biz.tin,
   330→      licenseNumber: biz.licenseNumber,
   331→      dateRegistered: biz.dateRegistered,
   332→      status: biz.status,
   333→      ghanaCard: biz.ghanaCard,
   334→      phone: biz.phone,
   335→      email: biz.email,
   336→      gpsAddress: biz.gpsAddress,
   337→      digitalAddress: biz.digitalAddress,
   338→      residentialAddress: biz.residentialAddress,
   339→      businessAddress: biz.businessAddress,
   340→      ward: biz.ward,
   341→      electoralArea: biz.electoralArea,
   342→      zone: biz.zone,
   343→      revenueArea: biz.revenueArea,
   344→      streetName: (biz as Record<string, unknown>).streetName as string || '',
   345→      houseNo: (biz as Record<string, unknown>).houseNo as string || '',
   346→      streetCode: (biz as Record<string, unknown>).streetCode as string || '',
   347→      localityCode: (biz as Record<string, unknown>).localityCode as string || '',
   348→      daAssignmentNo: (biz as Record<string, unknown>).daAssignmentNo as string || '',
   349→      businessCertNo: (biz as Record<string, unknown>).businessCertNo as string || '',
   350→      businessPermit: (biz as Record<string, unknown>).businessPermit as string || '',
   351→      employees: (biz as Record<string, unknown>).employees as string || '',
   352→      yearEstablished: (biz as Record<string, unknown>).yearEstablished as string || '',
   353→      excludedFromFees: (biz as Record<string, unknown>).excludedFromFees as boolean || false,
   354→      ownerAddress: (biz as Record<string, unknown>).ownerAddress as string || '',
   355→      ownerGps: (biz as Record<string, unknown>).ownerGps as string || '',
   356→      ownerTin: (biz as Record<string, unknown>).ownerTin as string || '',
   357→      comments: (biz as Record<string, unknown>).comments as string || '',
   358→    });
   359→    setView('form');
   360→  };
   361→
   362→  const handleDelete = (regNumber: string) => {
   363→    setBusinesses((prev) => prev.filter((b) => b.regNumber !== regNumber));
   364→  };
   365→
   366→  // ── Form Field Helper ────────────────────────────────────────────────────
   367→  const inputClass =
   368→    'w-full rounded-lg border border-slate-300 dark:border-slate-600 bg-white dark:bg-slate-800 px-4 py-2.5 text-sm focus:ring-2 focus:ring-emerald-500 focus:border-emerald-500 outline-none transition';
   369→  const labelClass =
   370→    'text-sm font-medium text-slate-700 dark:text-slate-300 mb-1.5';
   371→  const sectionHeaderClass =
   372→    'text-lg font-semibold text-slate-900 dark:text-white mb-4 pb-2 border-b border-slate-200 dark:border-slate-700';
   373→
   374→  // ══════════════════════════════════════════════════════════════════════════
   375→  //  LIST VIEW
   376→  // ══════════════════════════════════════════════════════════════════════════
   377→  if (view === 'list') {
   378→    return (
   379→      <div className="space-y-6">
   380→        {/* ── Header ──────────────────────────────────────────────────────── */}
   381→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   382→          <div>
   383→            <h1 className="text-2xl font-bold text-slate-900 dark:text-white">
   384→              Business Registration
   385→            </h1>
   386→            <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">
   387→              Manage and register businesses within the assembly. Track revenue
   388→              collection and compliance.
   389→            </p>
   390→          </div>
   391→          <button
   392→            onClick={() => { setEditingRegNumber(null); setForm({ ...defaultForm }); setView('form'); }}
   393→            className="inline-flex items-center gap-2 bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-medium px-4 py-2.5 rounded-lg transition-colors whitespace-nowrap"
   394→          >
   395→            <Plus className="w-4 h-4" />
   396→            Register New Business
   397→          </button>
   398→        </div>
   399→
   400→        {/* ── Search & Filters ────────────────────────────────────────────── */}
   401→        <div className="flex flex-col sm:flex-row gap-3">
   402→          <div className="relative flex-1">
   403→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   404→            <input
   405→              type="text"
   406→              placeholder="Search by name, owner, or registration number..."
   407→              value={searchQuery}
   408→              onChange={(e) => { setSearchQuery(e.target.value); setCurrentPage(1); }}
   409→              className={`${inputClass} pl-10`}
   410→            />
   411→          </div>
   412→          <select
   413→            value={statusFilter}
   414→            onChange={(e) => { setStatusFilter(e.target.value); setCurrentPage(1); }}
   415→            className={`${inputClass} w-full sm:w-44`}
   416→          >
   417→            <option value="All">All Statuses</option>
   418→            <option value="Active">Active</option>
   419→            <option value="Inactive">Inactive</option>
   420→          </select>
   421→          <select
   422→            value={typeFilter}
   423→            onChange={(e) => { setTypeFilter(e.target.value); setCurrentPage(1); }}
   424→            className={`${inputClass} w-full sm:w-48`}
   425→          >
   426→            <option value="All">All Business Classes</option>
   427→            {businessTypes.map((t) => (
   428→              <option key={t} value={t}>{t}</option>
   429→            ))}
   430→          </select>
   431→        </div>
   432→
   433→        {/* ── Table ───────────────────────────────────────────────────────── */}
   434→        <div className="rounded-xl border border-slate-200 dark:border-slate-700 overflow-hidden">
   435→          <div className="overflow-x-auto">
   436→            <table className="w-full text-sm">
   437→              <thead>
   438→                <tr className="bg-slate-50 dark:bg-slate-800/60 border-b border-slate-200 dark:border-slate-700">
   439→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Reg #</th>
   440→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Business Name</th>
   441→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Owner</th>
   442→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Business Class</th>
   443→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap hidden lg:table-cell">Category</th>
   444→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap hidden md:table-cell">TIN</th>
   445→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Status</th>
   446→                  <th className="text-right px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Actions</th>
   447→                </tr>
   448→              </thead>
   449→              <tbody className="divide-y divide-slate-100 dark:divide-slate-700">
   450→                {paged.length === 0 ? (
   451→                  <tr>
   452→                    <td colSpan={8} className="text-center py-12 text-slate-400 dark:text-slate-500">
   453→                      <Building2 className="w-10 h-10 mx-auto mb-3 opacity-40" />
   454→                      No businesses found. Click "Register New Business" to add one.
   455→                    </td>
   456→                  </tr>
   457→                ) : (
   458→                  paged.map((biz) => (
   459→                    <tr key={biz.regNumber} className="hover:bg-slate-50 dark:hover:bg-slate-800/40 transition-colors">
   460→                      <td className="px-4 py-3 font-mono text-xs text-slate-500 dark:text-slate-400 whitespace-nowrap">{biz.regNumber}</td>
   461→                      <td className="px-4 py-3 font-medium text-slate-900 dark:text-white whitespace-nowrap">{biz.name}</td>
   462→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap">{biz.owner}</td>
   463→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap">{biz.type}</td>
   464→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap hidden lg:table-cell">{biz.category}</td>
   465→                      <td className="px-4 py-3 font-mono text-xs text-slate-500 dark:text-slate-400 whitespace-nowrap hidden md:table-cell">{biz.tin}</td>
   466→                      <td className="px-4 py-3 whitespace-nowrap">
   467→                        <span className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-semibold ${
   468→                          biz.status === 'Active'
   469→                            ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400'
   470→                            : 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400'
   471→                        }`}>
   472→                          {biz.status}
   473→                        </span>
   474→                      </td>
   475→                      <td className="px-4 py-3 text-right whitespace-nowrap">
   476→                        <div className="inline-flex items-center gap-1">
   477→                          <button onClick={() => handleEdit(biz)} className="p-1.5 rounded-md text-slate-400 hover:text-emerald-600 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors" title="Edit">
   478→                            <Pencil className="w-4 h-4" />
   479→                          </button>
   480→                          <button onClick={() => handleDelete(biz.regNumber)} className="p-1.5 rounded-md text-slate-400 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-900/20 transition-colors" title="Delete">
   481→                            <Trash2 className="w-4 h-4" />
   482→                          </button>
   483→                        </div>
   484→                      </td>
   485→                    </tr>
   486→                  ))
   487→                )}
   488→              </tbody>
   489→            </table>
   490→          </div>
   491→        </div>
   492→
   493→        {/* ── Pagination ──────────────────────────────────────────────────── */}
   494→        <div className="flex flex-col sm:flex-row items-center justify-between gap-3 text-sm">
