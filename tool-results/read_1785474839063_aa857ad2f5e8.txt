   150→export function RentPage() {
   151→  const [view, setView] = useState<'list' | 'form'>('list');
   152→  const [searchQuery, setSearchQuery] = useState('');
   153→  const [currentPage, setCurrentPage] = useState(1);
   154→  const [rents, setRents] = useLocalStorage<Rent[]>('rms-rents', mockRents);
   155→  const [editingId, setEditingId] = useState<string | null>(null);
   156→  const itemsPerPage = 10;
   157→
   158→  // ── Form State ───────────────────────────────────────────────────────────
   159→  const defaultForm = {
   160→    upn: '',
   161→    streetName: '',
   162→    houseNo: '',
   163→    streetCode: '',
   164→    neighbourhood: '',
   165→    floor: '',
   166→    doorNo: '',
   167→    rentObjectName: '',
   168→    rentCode: '',
   169→    rentClass: '',
   170→    rentCategory: '',
   171→    rentUnit: '',
   172→    rentValue: '',
   173→    vacant: 'No',
   174→    startDate: '',
   175→    endDate: '',
   176→    contractId: '',
   177→    contractValue: '',
   178→    area: '',
   179→    renterName: '',
   180→    renterAddress: '',
   181→    renterGps: '',
   182→    phone: '',
   183→    email: '',
   184→    tin: '',
   185→    nationalId: '',
   186→    excludedFromRenting: false,
   187→    comments: '',
   188→  };
   189→
   190→  const [form, setForm] = useState(defaultForm);
   191→  const [locatingRenter, setLocatingRenter] = useState(false);
   192→
   193→  // ── Geolocation ──────────────────────────────────────────────────────────
   194→  const fetchGpsFromLocation = () => {
   195→    if (!navigator.geolocation) {
   196→      alert('Geolocation is not supported by your browser. Please enter the GPS address manually.');
   197→      return;
   198→    }
   199→    setLocatingRenter(true);
   200→    navigator.geolocation.getCurrentPosition(
   201→      (position) => {
   202→        const { latitude, longitude } = position.coords;
   203→        const gpsString = `${latitude.toFixed(6)}, ${longitude.toFixed(6)}`;
   204→        setForm((prev) => ({ ...prev, renterGps: gpsString }));
   205→        setLocatingRenter(false);
   206→      },
   207→      (error) => {
   208→        let message = 'Unable to retrieve location.';
   209→        switch (error.code) {
   210→          case error.PERMISSION_DENIED:
   211→            message = 'Location permission denied. Please allow location access in your browser settings.';
   212→            break;
   213→          case error.POSITION_UNAVAILABLE:
   214→            message = 'Location information is unavailable. Please try again or enter manually.';
   215→            break;
   216→          case error.TIMEOUT:
   217→            message = 'Location request timed out. Please try again.';
   218→            break;
   219→        }
   220→        alert(message);
   221→        setLocatingRenter(false);
   222→      },
   223→      { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
   224→    );
   225→  };
   226→

   228→  const filtered = rents.filter((r) => {
   229→    const q = searchQuery.toLowerCase();
   230→    return (
   231→      r.renterName.toLowerCase().includes(q) ||
   232→      r.upn.toLowerCase().includes(q) ||
   233→      r.streetName.toLowerCase().includes(q) ||
   234→      r.rentObjectName.toLowerCase().includes(q) ||
   235→      r.rentClass.toLowerCase().includes(q) ||
   236→      r.contractId.toLowerCase().includes(q)
   237→    );
   238→  });
   239→
   240→  const totalPages = Math.max(1, Math.ceil(filtered.length / itemsPerPage));
   241→  const startIdx = (currentPage - 1) * itemsPerPage;
   242→  const paged = filtered.slice(startIdx, startIdx + itemsPerPage);
   243→  const showingFrom = filtered.length === 0 ? 0 : startIdx + 1;
   244→  const showingTo = Math.min(startIdx + itemsPerPage, filtered.length);
   245→
   246→  // ── Handlers ─────────────────────────────────────────────────────────────
   247→  const handleFormChange = (
   248→    e: React.ChangeEvent<
   249→      HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement
   250→    >
   251→  ) => {
   252→    const { name, value, type } = e.target;
   253→    if (type === 'checkbox') {
   254→      setForm((prev) => ({ ...prev, [name]: (e.target as HTMLInputElement).checked }));
   255→    } else if (name === 'rentClass') {
   256→      // Auto-fill category: if only one option, select it; otherwise reset
   257→      const cats = RENT_CLASS_CATEGORIES[value] || [];
   258→      const autoCat = cats.length === 1 ? cats[0] : '';
   259→      const autoCode = autoCat ? (RENT_CODE_MAP[`${value}|${autoCat}`] || '') : '';
   260→      setForm((prev) => ({ ...prev, rentClass: value, rentCategory: autoCat, rentCode: autoCode }));
   261→    } else if (name === 'rentCategory') {
   262→      // Auto-fill code when category is selected
   263→      const code = RENT_CODE_MAP[`${form.rentClass}|${value}`] || '';
   264→      setForm((prev) => ({ ...prev, rentCategory: value, rentCode: code }));
   265→    } else {
   266→      setForm((prev) => ({ ...prev, [name]: value }));
   267→    }
   268→  };
   269→
   270→  const handleSave = () => {
   271→    if (!form.upn || !form.renterName) return;
   272→    if (editingId) {
   273→      setRents((prev) =>
   274→        prev.map((r) => (r.id === editingId ? { ...r, ...form } : r))
   275→      );
   276→      setEditingId(null);
   277→    } else {
   278→      const newRent: Rent = {
   279→        id: `RNT-${Date.now()}`,
   280→        ...form,
   281→      };
   282→      setRents((prev) => [...prev, newRent]);
   283→    }
   284→    setForm(defaultForm);
   285→    setView('list');
   286→  };
   287→
   288→  const handleEdit = (rent: Rent) => {
   289→    setForm({
   290→      upn: rent.upn,
   291→      streetName: rent.streetName,
   292→      houseNo: rent.houseNo,
   293→      streetCode: rent.streetCode,
   294→      neighbourhood: rent.neighbourhood,
   295→      floor: rent.floor,
   296→      doorNo: rent.doorNo,
   297→      rentObjectName: rent.rentObjectName,
   298→      rentCode: rent.rentCode || '',
   299→      rentClass: rent.rentClass,
   300→      rentCategory: rent.rentCategory || (RENT_CLASS_CATEGORIES[rent.rentClass]?.[0]) || '',
   301→      rentUnit: rent.rentUnit,
   302→      rentValue: rent.rentValue,
   303→      vacant: rent.vacant,
   304→      startDate: rent.startDate,
   305→      endDate: rent.endDate,
   306→      contractId: rent.contractId,
   307→      contractValue: rent.contractValue,
   308→      area: rent.area,
   309→      renterName: rent.renterName,
   310→      renterAddress: rent.renterAddress,
   311→      renterGps: rent.renterGps,
   312→      phone: rent.phone,
   313→      email: rent.email,
   314→      tin: rent.tin,
   315→      nationalId: rent.nationalId,
   316→      excludedFromRenting: rent.excludedFromRenting,
   317→      comments: rent.comments,
   318→    });
   319→    setEditingId(rent.id);
   320→    setView('form');
   321→  };
   322→
   323→  const handleDelete = (id: string) => {
   324→    if (confirm('Are you sure you want to delete this rent record?')) {
   325→      setRents((prev) => prev.filter((r) => r.id !== id));
   326→    }
   327→  };
   328→
   329→  const handleCancel = () => {
   330→    setForm(defaultForm);
   331→    setEditingId(null);
   332→    setView('list');
   333→  };
   334→
   335→  // ── Shared classes ──────────────────────────────────────────────────────
   336→  const inputClass =
   337→    'w-full rounded-lg border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-800 text-slate-900 dark:text-slate-100 px-3 py-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500 focus:border-emerald-500 transition-colors placeholder:text-slate-400 dark:placeholder:text-slate-500';
   338→  const labelClass =
   339→    'block text-xs font-medium text-slate-600 dark:text-slate-400 mb-1.5';
   340→  const cardClass =
   341→    'bg-white dark:bg-slate-800 rounded-xl border border-slate-200 dark:border-slate-700 p-4 sm:p-6 mb-4';
   342→  const cardHeaderClass =
   343→    'flex items-center gap-3 mb-4 pb-3 border-b border-slate-100 dark:border-slate-700';
   344→  const cardBodyClass = 'space-y-4';
   345→
   346→  // ── List View ─────────────────────────────────────────────────────────────
   347→  if (view === 'list') {
   348→    return (
   349→      <div className="min-h-screen bg-slate-50 dark:bg-slate-900">
   350→        <header className="bg-white dark:bg-slate-800 border-b border-slate-200 dark:border-slate-700 px-6 py-4">
   351→          <div className="max-w-7xl mx-auto flex items-center justify-between">
   352→            <h1 className="text-xl font-bold text-slate-900 dark:text-white">Rent Management</h1>
   353→            <button
   354→              onClick={() => { setForm(defaultForm); setEditingId(null); setView('form'); }}
   355→              className="inline-flex items-center gap-2 px-4 py-2.5 rounded-lg bg-emerald-600 text-white text-sm font-medium hover:bg-emerald-700 transition-colors"
   356→            >
   357→              <Plus className="w-4 h-4" /> Add Rent
   358→            </button>
   359→          </div>
   360→        </header>
   361→
   362→        <div className="max-w-7xl mx-auto px-6 py-6">

   364→          <div className="mb-4">

   367→              <input
   368→                type="text"
   369→                placeholder="Search by renter, UPN, street, rent object, class..."
   370→                value={searchQuery}
   371→                onChange={(e) => { setSearchQuery(e.target.value); setCurrentPage(1); }}
   372→                className={`${inputClass} pl-10`}
   373→              />
   374→            </div>
   375→          </div>
   376→
   377→          {/* Table */}
   378→          <div className="bg-white dark:bg-slate-800 rounded-xl border border-slate-200 dark:border-slate-700 overflow-hidden">
   379→            <div className="overflow-x-auto">
   380→              <table className="w-full text-left">
   381→                <thead className="bg-slate-50 dark:bg-slate-800/80 border-b border-slate-200 dark:border-slate-700">
   382→                  <tr>
   383→                    <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">UPN</th>
   384→                    <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Location</th>
   385→                    <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Rent Object</th>
   386→                    <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Renter</th>
   387→                    <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Rent Value</th>
   388→                    <th className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium px-4 py-3">Vacant</th>

   391→                  </tr>
   392→                </thead>
   393→                <tbody className="divide-y divide-slate-100 dark:divide-slate-700/60">
   394→                  {paged.length === 0 ? (
   395→                    <tr>
   396→                      <td colSpan={8} className="text-center py-12 text-slate-500 dark:text-slate-400">
   397→                        {searchQuery ? 'No rents match your search.' : 'No rents recorded yet. Click "Add Rent" to create one.'}
   398→                      </td>
   399→                    </tr>
   400→                  ) : (
   401→                    paged.map((rent) => (
   402→                      <tr key={rent.id} className="hover:bg-slate-50 dark:hover:bg-slate-800/40 transition-colors">
   403→                        <td className="px-4 py-3 font-mono text-xs text-slate-500 dark:text-slate-400 whitespace-nowrap">{rent.upn}</td>
   404→                        <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap">{rent.streetName}, {rent.houseNo}</td>
   405→                        <td className="px-4 py-3 font-medium text-slate-900 dark:text-white whitespace-nowrap">{rent.rentObjectName || '--NA--'}</td>
   406→                        <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap">{rent.renterName}</td>
   407→                        <td className="px-4 py-3 text-right font-medium text-slate-700 dark:text-slate-200 whitespace-nowrap">{rent.rentValue ? `GHS ${Number(rent.rentValue).toLocaleString()}` : '-'}</td>
   408→                        <td className="px-4 py-3 whitespace-nowrap">
   409→                          <span className={`inline-flex text-xs font-medium px-2 py-0.5 rounded-full ${
   410→                            rent.vacant === 'Yes'
   411→                              ? 'bg-amber-50 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400'
   412→                              : 'bg-emerald-50 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   413→                          }`}>
   414→                            {rent.vacant || 'No'}
   415→                          </span>
   416→                        </td>
   417→                        <td className="px-4 py-3 whitespace-nowrap">

   419→                            rent.excludedFromRenting
   420→                              ? 'bg-red-50 text-red-700 dark:bg-red-900/30 dark:text-red-400'
   421→                              : 'bg-emerald-50 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   422→                          }`}>
   423→                            {rent.excludedFromRenting ? 'Excluded' : 'Active'}
   424→                          </span>
   425→                        </td>
   426→                        <td className="px-4 py-3 whitespace-nowrap">
   427→                          <div className="flex items-center gap-1">

   433→                            </button>
   434→                          </div>
   435→                        </td>
   436→                      </tr>
   437→                    ))
   438→                  )}
   439→                </tbody>
   440→              </table>
   441→            </div>
   442→
   443→            {/* Pagination */}
   444→            {totalPages > 1 && (
   445→              <div className="flex items-center justify-between px-4 py-3 border-t border-slate-200 dark:border-slate-700">
   446→                <span className="text-sm text-slate-500 dark:text-slate-400">
   447→                  Showing {showingFrom}–{showingTo} of {filtered.length}
   448→                </span>
   449→                <div className="flex items-center gap-1">
   450→                  <button onClick={() => setCurrentPage((p) => Math.max(1, p - 1))} disabled={currentPage === 1} className="p-1.5 rounded-lg text-slate-400 hover:text-slate-600 disabled:opacity-30 transition-colors">
   451→                    <ChevronLeft className="w-4 h-4" />
   452→                  </button>
   453→                  <span className="text-sm text-slate-600 dark:text-slate-300 px-2">{currentPage} / {totalPages}</span>
   454→                  <button onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))} disabled={currentPage === totalPages} className="p-1.5 rounded-lg text-slate-400 hover:text-slate-600 disabled:opacity-30 transition-colors">
   455→                    <ChevronRight className="w-4 h-4" />
   456→                  </button>
   457→                </div>
   458→              </div>
   459→            )}
   460→          </div>
   461→        </div>
   462→      </div>
   463→    );
   464→  }
   465→
   466→  // ── Form View ─────────────────────────────────────────────────────────────
   467→  return (
   468→    <div className="min-h-screen bg-slate-50 dark:bg-slate-900">
   469→      <header className="bg-white dark:bg-slate-800 border-b border-slate-200 dark:border-slate-700 px-6 py-4">
   470→        <div className="max-w-7xl mx-auto flex items-center justify-between">
   471→          <div className="flex items-center gap-3">
   472→            <button onClick={handleCancel} className="p-2 rounded-lg text-slate-400 hover:text-slate-600 hover:bg-slate-100 dark:hover:bg-slate-700 transition-colors">
   473→              <ArrowLeft className="w-5 h-5" />
   474→            </button>
   475→            <h1 className="text-xl font-bold text-slate-900 dark:text-white">
   476→              {editingId ? 'Edit Rent' : 'Add Rent'}
   477→              {form.upn ? ` — UPN: ${form.upn}` : ''}
