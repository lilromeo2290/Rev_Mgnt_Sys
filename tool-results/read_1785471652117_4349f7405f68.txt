     1→'use client';
     2→
     3→import { useState } from 'react';
     4→import { useLocalStorage } from '@/hooks/use-local-storage';
     5→import {
     6→  Search,
     7→  Plus,
     8→  ArrowLeft,
     9→  Pencil,
    10→  Trash2,
    11→  ChevronLeft,
    12→  ChevronRight,
    13→  Home,
    14→  MapPin,
    15→  User,
    16→  Building2,
    17→  Crosshair,
    18→  Save,
    19→  X,
    20→  Loader2,
    21→} from 'lucide-react';
    22→
    23→// ─── Types ───────────────────────────────────────────────────────────────────
    24→

    26→  propNumber: string;
    27→  streetName: string;
    28→  houseNo: string;
    29→  streetCode: string;
    30→  ghanaPostGPS: string;
    31→  localityCode: string;
    32→  ownerName: string;
    33→  ownerAddress: string;
    34→  ownerGPS: string;
    35→  phone: string;
    36→  email: string;
    37→  tin: string;
    38→  nationalId: string;
    39→  ownershipType: string;
    40→  propertyUseType: string;
    41→  category: string;
    42→  value: string;
    43→  rooms: string;
    44→  hasBuildingPermit: string;
    45→  permitNumber: string;
    46→  excludedFromRating: boolean;
    47→  comments: string;
    48→}
    49→

    51→
    52→const mockProperties: Property[] = [];
    53→
    54→// ─── Constants ──────────────────────────────────────────────────────────────
    55→
    56→const OWNERSHIP_TYPES = [
    57→  'Private Individual',
    58→  'Private Company',
    59→  'Government',
    60→  'State Enterprise',
    61→  'Religious Body',
    62→  'Traditional Authority',
    63→  'Joint Ownership',
    64→  'Other',
    65→];
    66→
    67→const propertyUseTypes = [
    68→  '20121 : Residential : 3rd Class Residential : 420',
    69→  '20122 : Residential : 2nd Class Residential : 520',
    70→  '20123 : Residential : 1st Class Residential : 750',
    71→  '20201 : Commercial : 3rd Class Commercial : 650',
    72→  '20202 : Commercial : 2nd Class Commercial : 850',
    73→  '20203 : Commercial : 1st Class Commercial : 1200',
    74→  '20301 : Industrial : Light Industrial : 500',
    75→  '20302 : Industrial : Heavy Industrial : 800',
    76→  '20401 : Institutional : Educational : 400',
    77→  '20402 : Institutional : Health : 450',
    78→  '20501 : Mixed Use : Residential-Commercial : 900',
    79→];
    80→
    81→const propertyTypes = ['All', 'Residential', 'Commercial', 'Industrial', 'Institutional', 'Mixed Use'];
    82→const occupancyStatuses = ['All', 'Occupied', 'Vacant', 'Under Construction'];
    83→
    84→// ─── Component ───────────────────────────────────────────────────────────────
    85→
    86→export function PropertiesPage() {
    87→  const [view, setView] = useState<'list' | 'form'>('list');
    88→  const [searchQuery, setSearchQuery] = useState('');
    89→  const [typeFilter, setTypeFilter] = useState<string>('All');
    90→  const [statusFilter, setStatusFilter] = useState<string>('All');
    91→  const [currentPage, setCurrentPage] = useState(1);
    92→  const [properties, setProperties] = useLocalStorage<Property[]>('rms-properties', mockProperties);
    93→  const itemsPerPage = 10;
    94→
    95→  // ── Form State ───────────────────────────────────────────────────────────
    96→  const defaultForm = {
    97→    propNumber: '',
    98→    streetName: '',
    99→    houseNo: '',
   100→    streetCode: '',
   101→    ghanaPostGPS: '',
   102→    localityCode: '',
   103→    ownerName: '',
   104→    ownerAddress: '',
   105→    ownerGPS: '',
   106→    phone: '',
   107→    email: '',
   108→    tin: '',
   109→    nationalId: '',
   110→    ownershipType: '',
   111→    propertyUseType: '',
   112→    category: '',
   113→    value: '',
   114→    rooms: '',
   115→    hasBuildingPermit: 'No',
   116→    permitNumber: '',
   117→    excludedFromRating: false,
   118→    comments: '',
   119→  };
   120→
   121→  const [form, setForm] = useState(defaultForm);
   122→  const [locatingProperty, setLocatingProperty] = useState(false);
   123→  const [locatingOwner, setLocatingOwner] = useState(false);
   124→
   125→  // ── Geolocation: Fetch GPS from device location ─────────────────────
   126→  const fetchGpsFromLocation = (
   127→    target: 'property' | 'owner'
   128→  ) => {
   129→    const setLoading = target === 'property' ? setLocatingProperty : setLocatingOwner;
   130→
   131→    if (!navigator.geolocation) {
   132→      alert('Geolocation is not supported by your browser. Please enter the GPS address manually.');
   133→      return;
   134→    }
   135→
   136→    setLoading(true);
   137→
   138→    navigator.geolocation.getCurrentPosition(
   139→      (position) => {
   140→        const { latitude, longitude } = position.coords;
   141→        const gpsString = `${latitude.toFixed(6)}, ${longitude.toFixed(6)}`;
   142→        if (target === 'property') {
   143→          setForm((prev) => ({ ...prev, ghanaPostGPS: gpsString }));
   144→        } else {
   145→          setForm((prev) => ({ ...prev, ownerGPS: gpsString }));
   146→        }
   147→        setLoading(false);
   148→      },
   149→      (error) => {
   150→        let message = 'Unable to retrieve location.';
   151→        switch (error.code) {
   152→          case error.PERMISSION_DENIED:
   153→            message = 'Location permission denied. Please allow location access in your browser settings.';
   154→            break;
   155→          case error.POSITION_UNAVAILABLE:
   156→            message = 'Location information is unavailable. Please try again or enter manually.';
   157→            break;
   158→          case error.TIMEOUT:
   159→            message = 'Location request timed out. Please try again.';
   160→            break;
   161→        }
   162→        alert(message);
   163→        setLoading(false);
   164→      },
   165→      { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
   166→    );
   167→  };
   168→
   169→  // ── Filtering ─────────────────────────────────────────────────────────────
   170→  const filtered = properties.filter((p) => {
   171→    const matchSearch =
   172→      searchQuery === '' ||
   173→      p.ownerName.toLowerCase().includes(searchQuery.toLowerCase()) ||
   174→      p.propNumber.toLowerCase().includes(searchQuery.toLowerCase()) ||
   175→      p.streetName.toLowerCase().includes(searchQuery.toLowerCase());
   176→    const matchType = typeFilter === 'All' || p.propertyUseType.toLowerCase().includes(typeFilter.toLowerCase());
   177→    return matchSearch && matchType;
   178→  });
   179→
   180→  const totalPages = Math.max(1, Math.ceil(filtered.length / itemsPerPage));
   181→  const safeCurrentPage = Math.min(currentPage, totalPages);
   182→  const startIdx = (safeCurrentPage - 1) * itemsPerPage;
   183→  const paged = filtered.slice(startIdx, startIdx + itemsPerPage);
   184→  const showingFrom = filtered.length === 0 ? 0 : startIdx + 1;
   185→  const showingTo = Math.min(startIdx + itemsPerPage, filtered.length);
   186→
   187→  // ── Handlers ─────────────────────────────────────────────────────────────
   188→  // Derived: unique categories from propertyUseTypes
   189→  const categories = [...new Set(propertyUseTypes.map((t) => t.split(':')[1]?.trim()).filter(Boolean))];
   190→
   191→  // Derived: sub-categories for the selected category
   192→  const subCategories = form.category
   193→    ? propertyUseTypes.filter((t) => t.split(':')[1]?.trim() === form.category)
   194→    : [];
   195→
   196→  const handleFormChange = (
   197→    e: React.ChangeEvent<
   198→      HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement
   199→    >
   200→  ) => {
   201→    const { name, value, type } = e.target;
   202→    if (type === 'checkbox') {
   203→      setForm((prev) => ({ ...prev, [name]: (e.target as HTMLInputElement).checked }));
   204→    } else if (name === 'category') {
   205→      // When category changes, reset propertyUseType so user picks a sub-category
   206→      setForm((prev) => ({ ...prev, category: value, propertyUseType: '' }));
   207→    } else if (name === 'propertyUseType') {
   208→      // When propertyUseType changes, auto-fill category
   209→      const cat = value ? value.split(':')[1]?.trim() || '' : '';
   210→      setForm((prev) => ({ ...prev, [name]: value, category: cat }));
   211→    } else {
   212→      setForm((prev) => ({ ...prev, [name]: value }));
   213→    }
   214→  };
   215→
   216→  const handleSave = () => {
   217→    if (!form.ownerName || !form.propertyUseType) return;
   218→    const newProp: Property = {
   219→      propNumber: form.propNumber || `UPN-${String(properties.length + 1).padStart(4, '0')}`,
   220→      ...form,
   221→    };
   222→    setProperties((prev) => [...prev, newProp]);
   223→    setForm({ ...defaultForm });
   224→    setView('list');
   225→  };
   226→
   227→  const handleCancel = () => {
   228→    setView('list');
   229→  };
   230→
   231→  const handleDelete = (propNumber: string) => {
   232→    setProperties((prev) => prev.filter((p) => p.propNumber !== propNumber));
   233→  };
   234→
   235→  // ── Form Helpers ─────────────────────────────────────────────────────────
   236→  const inputClass =
   237→    'w-full rounded-lg border border-slate-300 dark:border-slate-600 bg-white dark:bg-slate-800 px-4 py-2.5 text-sm focus:ring-2 focus:ring-emerald-500 focus:border-emerald-500 outline-none transition';
   238→  const labelClass =
   239→    'block text-sm font-medium text-slate-700 dark:text-slate-300 mb-1';
   240→
   241→  const cardClass = 'rounded-xl border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900 overflow-hidden';
   242→  const cardHeaderClass = 'flex items-center gap-2.5 px-5 py-3.5 bg-slate-50 dark:bg-slate-800/60 border-b border-slate-200 dark:border-slate-700';
   243→  const cardBodyClass = 'p-5 space-y-4';
   244→
   245→  const statusBadge = (status: string) => {
   246→    const styles: Record<string, string> = {
   247→      Occupied: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400',
   248→      Vacant: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400',
   249→      'Under Construction': 'bg-blue-100 text-blue-700 dark:bg-blue-900/40 dark:text-blue-400',
   250→    };
   251→    return (
   252→      <span className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-semibold ${styles[status] || 'bg-slate-100 text-slate-600'}`}>
   253→        {status}
   254→      </span>
   255→    );
   256→  };
   257→
   258→  const formatVal = (v: string) => {
   259→    const n = parseFloat(v);
   260→    if (isNaN(n)) return 'GH₵ 0';
   261→    return `GH\u20b5 ${n.toLocaleString('en-GH')}`;
   262→  };
   263→
   264→  // ══════════════════════════════════════════════════════════════════════════
   265→  //  LIST VIEW
   266→  // ══════════════════════════════════════════════════════════════════════════
   267→  if (view === 'list') {
   268→    return (
   269→      <div className="space-y-6">
   270→        {/* ── Header ──────────────────────────────────────────────────────── */}
   271→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   272→          <div>
   273→            <h1 className="text-2xl font-bold text-slate-900 dark:text-white">Property Registration</h1>
   274→            <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">
   275→              Manage and register properties within the assembly.
   276→            </p>
   277→          </div>
   278→          <button
   279→            onClick={() => setView('form')}
   280→            className="inline-flex items-center gap-2 bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-medium px-4 py-2.5 rounded-lg transition-colors whitespace-nowrap cursor-pointer"
   281→          >
   282→            <Plus className="w-4 h-4" />
   283→            Register New Property
   284→          </button>
   285→        </div>
   286→
   287→        {/* ── Search & Filters ────────────────────────────────────────────── */}
   288→        <div className="flex flex-col sm:flex-row gap-3">
   289→          <div className="relative flex-1">
   290→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   291→            <input
   292→              type="text"
   293→              placeholder="Search by UPN, owner, or street..."
   294→              value={searchQuery}
   295→              onChange={(e) => { setSearchQuery(e.target.value); setCurrentPage(1); }}
   296→              className={`${inputClass} pl-10`}
   297→            />
   298→          </div>
   299→          <select
   300→            value={typeFilter}
   301→            onChange={(e) => { setTypeFilter(e.target.value); setCurrentPage(1); }}
   302→            className={`${inputClass} w-full sm:w-48`}
   303→          >
   304→            {propertyTypes.map((t) => (
   305→              <option key={t} value={t}>{t === 'All' ? 'All Property Types' : t}</option>
   306→            ))}
   307→          </select>
   308→          <select
   309→            value={statusFilter}
   310→            onChange={(e) => { setStatusFilter(e.target.value); setCurrentPage(1); }}
   311→            className={`${inputClass} w-full sm:w-48`}
   312→          >
   313→            {occupancyStatuses.map((s) => (
   314→              <option key={s} value={s}>{s === 'All' ? 'All Statuses' : s}</option>
   315→            ))}
   316→          </select>
   317→        </div>
   318→
   319→        {/* ── Table ───────────────────────────────────────────────────────── */}
   320→        <div className="rounded-xl border border-slate-200 dark:border-slate-700 overflow-hidden">
   321→          <div className="overflow-x-auto">
   322→            <table className="w-full text-sm">
   323→              <thead>
   324→                <tr className="bg-slate-50 dark:bg-slate-800/60 border-b border-slate-200 dark:border-slate-700">
   325→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">UPN</th>
   326→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Owner</th>
   327→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Property Use Type</th>
   328→                  <th className="text-right px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Value</th>
   329→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Street</th>
   330→                  <th className="text-right px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">Actions</th>
   331→                </tr>
   332→              </thead>
   333→              <tbody className="divide-y divide-slate-100 dark:divide-slate-700">
   334→                {paged.length === 0 ? (
   335→                  <tr>
   336→                    <td colSpan={6} className="text-center py-12 text-slate-400 dark:text-slate-500">
   337→                      <Home className="w-10 h-10 mx-auto mb-3 opacity-40" />
   338→                      No properties found.
   339→                    </td>
   340→                  </tr>
   341→                ) : (
   342→                  paged.map((prop) => (
   343→                    <tr key={prop.propNumber} className="hover:bg-slate-50 dark:hover:bg-slate-800/40 transition-colors">
   344→                      <td className="px-4 py-3 font-mono text-xs text-slate-500 dark:text-slate-400 whitespace-nowrap">{prop.propNumber}</td>
   345→                      <td className="px-4 py-3 font-medium text-slate-900 dark:text-white whitespace-nowrap">{prop.ownerName}</td>
   346→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap max-w-[200px] truncate">{(prop.propertyUseType || '').split(':')[1] ? (prop.propertyUseType || '').split(':')[1].trim() : (prop.propertyUseType || '')}</td>
   347→                      <td className="px-4 py-3 text-right font-medium text-slate-700 dark:text-slate-200 whitespace-nowrap">{formatVal(prop.value)}</td>
   348→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap">{prop.streetName}</td>
   349→                      <td className="px-4 py-3 text-right whitespace-nowrap">
   350→                        <div className="inline-flex items-center gap-1">
   351→                          <button className="p-1.5 rounded-md text-slate-400 hover:text-emerald-600 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors cursor-pointer" title="Edit"><Pencil className="w-4 h-4" /></button>
   352→                          <button onClick={() => handleDelete(prop.propNumber)} className="p-1.5 rounded-md text-slate-400 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-900/20 transition-colors cursor-pointer" title="Delete"><Trash2 className="w-4 h-4" /></button>
   353→                        </div>
   354→                      </td>
   355→                    </tr>
   356→                  ))
   357→                )}
   358→              </tbody>
   359→            </table>
   360→          </div>
   361→        </div>
   362→
   363→        {/* ── Pagination ──────────────────────────────────────────────────── */}
   364→        <div className="flex flex-col sm:flex-row items-center justify-between gap-3 text-sm">
   365→          <p className="text-slate-500 dark:text-slate-400">Showing {showingFrom}-{showingTo} of {filtered.length}</p>
   366→          <div className="flex items-center gap-2">
   367→            <button onClick={() => setCurrentPage((p) => Math.max(1, p - 1))} disabled={safeCurrentPage <= 1} className="inline-flex items-center gap-1 px-3 py-1.5 rounded-lg border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-800 disabled:opacity-40 disabled:cursor-not-allowed transition-colors cursor-pointer">
   368→              <ChevronLeft className="w-4 h-4" /> Previous
   369→            </button>
   370→            <span className="px-3 py-1.5 rounded-lg bg-slate-100 dark:bg-slate-800 text-slate-700 dark:text-slate-300 font-medium">{safeCurrentPage} / {totalPages}</span>
   371→            <button onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))} disabled={safeCurrentPage >= totalPages} className="inline-flex items-center gap-1 px-3 py-1.5 rounded-lg border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-800 disabled:opacity-40 disabled:cursor-not-allowed transition-colors cursor-pointer">
   372→              Next <ChevronRight className="w-4 h-4" />
   373→            </button>
   374→          </div>
   375→        </div>
   376→      </div>
   377→    );
   378→  }
   379→
   380→  // ══════════════════════════════════════════════════════════════════════════
   381→  //  FORM VIEW — 3-Card Layout
   382→  // ══════════════════════════════════════════════════════════════════════════
   383→  return (
   384→    <div className="space-y-6">
   385→      {/* ── Header ────────────────────────────────────────────────────────── */}
   386→      <div className="flex items-center gap-4">
   387→        <button onClick={handleCancel} className="p-2 rounded-lg border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-800 transition-colors cursor-pointer">
   388→          <ArrowLeft className="w-5 h-5" />
   389→        </button>
   390→        <div>
   391→          <h1 className="text-2xl font-bold text-slate-900 dark:text-white">Register New Property</h1>
   392→          <p className="text-sm text-slate-500 dark:text-slate-400 mt-0.5">Fill in the details below to register a new property.</p>
   393→        </div>
   394→      </div>
   395→
   396→      <div className="space-y-5">
   397→        {/* ════════════════════════════════════════════════════════════════
   398→           CARD 1: LOCATION
   399→           ════════════════════════════════════════════════════════════════ */}
   400→        <div className={cardClass}>
   401→          <div className={cardHeaderClass}>
   402→            <div className="w-8 h-8 rounded-lg bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   403→              <MapPin className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   404→            </div>
   405→            <h2 className="text-sm font-semibold text-slate-800 dark:text-slate-200 uppercase tracking-wider">Location</h2>
   406→          </div>
   407→          <div className={cardBodyClass}>
   408→            {/* Row 1: Street Name (wider) | House No. | Street Code */}
   409→            <div className="grid grid-cols-12 gap-4">
   410→              <div className="col-span-12 sm:col-span-5">
   411→                <label className={labelClass}>Street Name</label>
   412→                <input type="text" name="streetName" value={form.streetName} onChange={handleFormChange} placeholder="Enter street name" className={inputClass} />
   413→              </div>
   414→              <div className="col-span-6 sm:col-span-3">
   415→                <label className={labelClass}>House No.</label>
   416→                <input type="text" name="houseNo" value={form.houseNo} onChange={handleFormChange} placeholder="e.g. 26" className={inputClass} />
   417→              </div>
   418→              <div className="col-span-6 sm:col-span-4">
   419→                <label className={labelClass}>Street Code</label>
   420→                <input type="text" name="streetCode" value={form.streetCode} onChange={handleFormChange} placeholder="Enter code" className={inputClass} />
   421→              </div>
   422→            </div>
   423→            {/* Row 2: GhanaPost GPS (with button) | Locality Code */}
   424→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   425→              <div>
   426→                <label className={labelClass}>GhanaPost GPS</label>
   427→                <div className="flex gap-2">
   428→                  <input type="text" name="ghanaPostGPS" value={form.ghanaPostGPS} onChange={handleFormChange} placeholder="e.g. AK-034-5521" className={`${inputClass} flex-1`} />
   429→                  <button type="button" onClick={() => fetchGpsFromLocation('property')} disabled={locatingProperty} className="inline-flex items-center gap-1.5 px-3 py-2.5 rounded-lg border border-emerald-300 dark:border-emerald-700 text-emerald-700 dark:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 disabled:opacity-50 transition-colors whitespace-nowrap" title="Use device GPS">
   430→                    {locatingProperty ? <Loader2 className="w-4 h-4 animate-spin" /> : <Crosshair className="w-4 h-4" />}
   431→                    {locatingProperty ? '...' : 'GPS'}
   432→                  </button>
   433→                </div>
   434→              </div>
   435→              <div>
   436→                <label className={labelClass}>Locality Code</label>
   437→                <input type="text" name="localityCode" value={form.localityCode} onChange={handleFormChange} placeholder="Enter locality code" className={inputClass} />
   438→              </div>
   439→            </div>
   440→          </div>
   441→        </div>
   442→
   443→        {/* ════════════════════════════════════════════════════════════════
   444→           CARD 2: OWNER INFORMATION
   445→           ════════════════════════════════════════════════════════════════ */}
   446→        <div className={cardClass}>
   447→          <div className={cardHeaderClass}>
