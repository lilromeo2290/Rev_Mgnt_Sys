   176→        switch (error.code) {
   177→          case error.PERMISSION_DENIED:
   178→            message = 'Location permission denied. Please allow location access in your browser settings.';
   179→            break;
   180→          case error.POSITION_UNAVAILABLE:
   181→            message = 'Location information is unavailable. Please try again or enter manually.';
   182→            break;
   183→          case error.TIMEOUT:
   184→            message = 'Location request timed out. Please try again.';

   188→        setLoading(false);
   189→      },
