   301→
   302→export function PropertiesPage() {
   303→  const [view, setView] = useState<'list' | 'form'>('list');
   304→  const [searchQuery, setSearchQuery] = useState('');
   305→  const [typeFilter, setTypeFilter] = useState<string>('All');
   306→  const [statusFilter, setStatusFilter] = useState<string>('All');
   307→  const [currentPage, setCurrentPage] = useState(1);
   308→  const [properties, setProperties] = useState<Property[]>(mockProperties);
   309→  const itemsPerPage = 10;
   310→
   311→  // ── Form State ───────────────────────────────────────────────────────────
   312→  const [form, setForm] = useState({
   313→    propNumber: '',
   314→    owner: '',
   315→    ownerContact: '',
   316→    propertyType: '',
   317→    category: '',
   318→    floors: '',
   319→    rooms: '',
   320→    propertyUse: '',
   321→    valuation: '',
   322→    occupancyStatus: 'Occupied',
   323→    buildingPermit: '',
   324→    gpsCoordinates: '',
   325→    digitalAddress: '',
   326→    plotNumber: '',
   327→    streetName: '',
   328→    ward: '',
   329→    electoralArea: '',
   330→    zone: '',
   331→  });
   332→
   333→  // ── Derived categories based on selected property type ──────────────────
   334→  const availableCategories = form.propertyType
   335→    ? propertyCategories[form.propertyType] || []
   336→    : [];
   337→
   338→  // ── Filtering & Pagination ───────────────────────────────────────────────
   339→  const filtered = properties.filter((p) => {
   340→    const matchSearch =
   341→      searchQuery === '' ||
   342→      p.owner.toLowerCase().includes(searchQuery.toLowerCase()) ||
   343→      p.propNumber.toLowerCase().includes(searchQuery.toLowerCase()) ||
   344→      p.streetName.toLowerCase().includes(searchQuery.toLowerCase());
   345→    const matchType = typeFilter === 'All' || p.propertyType === typeFilter;
   346→    const matchStatus =
   347→      statusFilter === 'All' || p.occupancyStatus === statusFilter;
   348→    return matchSearch && matchType && matchStatus;
   349→  });
   350→
   351→  const totalPages = Math.max(1, Math.ceil(filtered.length / itemsPerPage));
   352→  const safeCurrentPage = Math.min(currentPage, totalPages);
   353→  const startIdx = (safeCurrentPage - 1) * itemsPerPage;
   354→  const paged = filtered.slice(startIdx, startIdx + itemsPerPage);
   355→  const showingFrom = filtered.length === 0 ? 0 : startIdx + 1;
   356→  const showingTo = Math.min(startIdx + itemsPerPage, filtered.length);
   357→
   358→  // ── Handlers ─────────────────────────────────────────────────────────────
   359→  const handleFormChange = (
   360→    e: React.ChangeEvent<
   361→      HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement
   362→    >
   363→  ) => {
   364→    const { name, value } = e.target;
   365→    setForm((prev) => {
   366→      const updated = { ...prev, [name]: value };
   367→      if (name === 'propertyType') {
   368→        updated.category = '';
   369→      }
   370→      return updated;
   371→    });
   372→  };
   373→
   374→  const handleSave = () => {
   375→    setView('list');
   376→  };
   377→
   378→  const handleCancel = () => {
   379→    setView('list');
   380→  };
   381→
   382→  const handleDelete = (propNumber: string) => {
   383→    setProperties((prev) => prev.filter((p) => p.propNumber !== propNumber));
   384→  };
   385→
   386→  // ── Form Field Helper ────────────────────────────────────────────────────
   387→  const inputClass =
   388→    'w-full rounded-lg border border-slate-300 dark:border-slate-600 bg-white dark:bg-slate-800 px-4 py-2.5 text-sm focus:ring-2 focus:ring-emerald-500 focus:border-emerald-500 outline-none transition';
   389→  const labelClass =
   390→    'text-sm font-medium text-slate-700 dark:text-slate-300 mb-1.5';
   391→  const sectionHeaderClass =
   392→    'text-lg font-semibold text-slate-900 dark:text-white mb-4 pb-2 border-b border-slate-200 dark:border-slate-700';
   393→
   394→  // ── Status Badge Helper ─────────────────────────────────────────────────
   395→  const statusBadge = (status: Property['occupancyStatus']) => {
   396→    const styles: Record<
   397→      Property['occupancyStatus'],
   398→      string
   399→    > = {
   400→      Occupied:
   401→        'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400',
   402→      Vacant:
   403→        'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400',
   404→      'Under Construction':
   405→        'bg-blue-100 text-blue-700 dark:bg-blue-900/40 dark:text-blue-400',
   406→    };
   407→    return (
   408→      <span
   409→        className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-semibold ${styles[status]}`}
   410→      >
   411→        {status}
   412→      </span>
   413→    );
   414→  };
   415→
   416→  // ── Format Valuation ────────────────────────────────────────────────────
   417→  const formatValuation = (amount: number) =>
   418→    amount.toLocaleString('en-GH');
   419→
   420→  // ══════════════════════════════════════════════════════════════════════════
   421→  //  LIST VIEW
   422→  // ══════════════════════════════════════════════════════════════════════════
   423→  if (view === 'list') {
   424→    return (
   425→      <div className="space-y-6">
   426→        {/* ── Header ──────────────────────────────────────────────────────── */}
   427→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   428→          <div>
   429→            <h1 className="text-2xl font-bold text-slate-900 dark:text-white">
   430→              Property Registration
   431→            </h1>
   432→            <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">
   433→              Manage and register properties within the assembly. Track
   434→              valuations, occupancy, and rates collection.
   435→            </p>
   436→          </div>
   437→          <button
   438→            onClick={() => setView('form')}
   439→            className="inline-flex items-center gap-2 bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-medium px-4 py-2.5 rounded-lg transition-colors whitespace-nowrap"
   440→          >
   441→            <Plus className="w-4 h-4" />
   442→            Register New Property
   443→          </button>
   444→        </div>
   445→
   446→        {/* ── Search & Filters ────────────────────────────────────────────── */}
   447→        <div className="flex flex-col sm:flex-row gap-3">
   448→          <div className="relative flex-1">
   449→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   450→            <input
   451→              type="text"
   452→              placeholder="Search by property number, owner, or street..."
   453→              value={searchQuery}
   454→              onChange={(e) => {
   455→                setSearchQuery(e.target.value);
   456→                setCurrentPage(1);
   457→              }}
   458→              className={`${inputClass} pl-10`}
   459→            />
   460→          </div>
   461→          <select
   462→            value={typeFilter}
   463→            onChange={(e) => {
   464→              setTypeFilter(e.target.value);
   465→              setCurrentPage(1);
   466→            }}
   467→            className={`${inputClass} w-full sm:w-48`}
   468→          >
   469→            <option value="All">All Property Types</option>
   470→            {propertyTypes.map((t) => (
   471→              <option key={t} value={t}>
   472→                {t}
   473→              </option>
   474→            ))}
   475→          </select>
   476→          <select
   477→            value={statusFilter}
   478→            onChange={(e) => {
   479→              setStatusFilter(e.target.value);
   480→              setCurrentPage(1);
   481→            }}
   482→            className={`${inputClass} w-full sm:w-48`}
   483→          >
   484→            {occupancyStatuses.map((s) => (
   485→              <option key={s} value={s}>
   486→                {s === 'All' ? 'All Occupancy Statuses' : s}
   487→              </option>
   488→            ))}
   489→          </select>
   490→        </div>
   491→
   492→        {/* ── Table ───────────────────────────────────────────────────────── */}
   493→        <div className="rounded-xl border border-slate-200 dark:border-slate-700 overflow-hidden">
   494→          <div className="overflow-x-auto">
   495→            <table className="w-full text-sm">
   496→              <thead>
   497→                <tr className="bg-slate-50 dark:bg-slate-800/60 border-b border-slate-200 dark:border-slate-700">
   498→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   499→                    Property #
   500→                  </th>
   501→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   502→                    Owner
   503→                  </th>
   504→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   505→                    Property Type
   506→                  </th>
   507→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap hidden lg:table-cell">
   508→                    Category
   509→                  </th>
   510→                  <th className="text-right px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   511→                    Valuation
   512→                  </th>
   513→                  <th className="text-left px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   514→                    Status
   515→                  </th>
   516→                  <th className="text-right px-4 py-3 font-semibold text-slate-600 dark:text-slate-300 whitespace-nowrap">
   517→                    Actions
   518→                  </th>
   519→                </tr>
   520→              </thead>
   521→              <tbody className="divide-y divide-slate-100 dark:divide-slate-700">
   522→                {paged.length === 0 ? (
   523→                  <tr>
   524→                    <td
   525→                      colSpan={7}
   526→                      className="text-center py-12 text-slate-400 dark:text-slate-500"
   527→                    >
   528→                      <Home className="w-10 h-10 mx-auto mb-3 opacity-40" />
   529→                      No properties found matching your criteria.
   530→                    </td>
   531→                  </tr>
   532→                ) : (
   533→                  paged.map((prop) => (
   534→                    <tr
   535→                      key={prop.propNumber}
   536→                      className="hover:bg-slate-50 dark:hover:bg-slate-800/40 transition-colors"
   537→                    >
   538→                      <td className="px-4 py-3 font-mono text-xs text-slate-500 dark:text-slate-400 whitespace-nowrap">
   539→                        {prop.propNumber}
   540→                      </td>
   541→                      <td className="px-4 py-3 font-medium text-slate-900 dark:text-white whitespace-nowrap">
   542→                        {prop.owner}
   543→                      </td>
   544→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap">
   545→                        {prop.propertyType}
   546→                      </td>
   547→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap hidden lg:table-cell">
   548→                        {prop.category}
   549→                      </td>
   550→                      <td className="px-4 py-3 text-right font-medium text-slate-700 dark:text-slate-200 whitespace-nowrap">
   551→                        GH₵ {formatValuation(prop.valuation)}
   552→                      </td>
   553→                      <td className="px-4 py-3 whitespace-nowrap">
   554→                        {statusBadge(prop.occupancyStatus)}
   555→                      </td>
   556→                      <td className="px-4 py-3 text-right whitespace-nowrap">
   557→                        <div className="inline-flex items-center gap-1">
   558→                          <button
   559→                            className="p-1.5 rounded-md text-slate-400 hover:text-emerald-600 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   560→                            title="Edit"
   561→                          >
   562→                            <Pencil className="w-4 h-4" />
   563→                          </button>
