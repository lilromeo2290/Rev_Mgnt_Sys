   100→

   103→  const [searchQuery, setSearchQuery] = useState('');
   104→  const [typeFilter, setTypeFilter] = useState<string>('All');

   109→
   110→  // ── Form State ───────────────────────────────────────────────────────────

   113→    owner: '',
   114→    ownerContact: '',

   116→    category: '',
   117→    floors: '',
   118→    rooms: '',
   119→    propertyUse: '',
   120→    valuation: '',
   121→    occupancyStatus: 'Occupied',
   122→    buildingPermit: '',
   123→    gpsCoordinates: '',
   124→    digitalAddress: '',
   125→    plotNumber: '',
   126→    streetName: '',
   127→    ward: '',
   128→    electoralArea: '',
   129→    zone: '',

   154→  const showingFrom = filtered.length === 0 ? 0 : startIdx + 1;

   166→      if (name === 'propertyType') {
   167→        updated.category = '';

   170→    });

   174→    setView('list');

   177→  const handleCancel = () => {

   181→  const handleDelete = (propNumber: string) => {
   182→    setProperties((prev) => prev.filter((p) => p.propNumber !== propNumber));

   331→                ) : (
   332→                  paged.map((prop) => (

   341→                        {prop.owner}
   342→                      </td>
   343→                      <td className="px-4 py-3 text-slate-600 dark:text-slate-300 whitespace-nowrap">
   344→                        {prop.propertyType}
   345→                      </td>

   350→                        GH₵ {formatValuation(prop.valuation)}

   362→                          </button>

   366→                            title="Delete"
   367→                          >
   368→                            <Trash2 className="w-4 h-4" />
   369→                          </button>
   370→                        </div>
   371→                      </td>
   372→                    </tr>

   375→              </tbody>
   376→            </table>

   383→            Showing {showingFrom}-{showingTo} of {filtered.length}
   384→          </p>
   385→          <div className="flex items-center gap-2">
   386→            <button
   387→              onClick={() => setCurrentPage((p) => Math.max(1, p - 1))}
   388→              disabled={safeCurrentPage <= 1}
   389→              className="inline-flex items-center gap-1 px-3 py-1.5 rounded-lg border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-800 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   390→            >
   391→              <ChevronLeft className="w-4 h-4" />
   392→              Previous
   393→            </button>
   394→            <span className="px-3 py-1.5 rounded-lg bg-slate-100 dark:bg-slate-800 text-slate-700 dark:text-slate-300 font-medium">
   395→              {safeCurrentPage} / {totalPages}
   396→            </span>
   397→            <button
   398→              onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))}
   399→              disabled={safeCurrentPage >= totalPages}
   400→              className="inline-flex items-center gap-1 px-3 py-1.5 rounded-lg border border-slate-300 dark:border-slate-600 text-slate-600 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-800 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"

   402→              Next

   424→        <div>

   428→          <p className="text-sm text-slate-500 dark:text-slate-400 mt-0.5">
   429→            Fill in the details below to register a new property.
   430→          </p>
   431→        </div>
   432→      </div>
   433→
   434→      <div className="space-y-8">
   435→        {/* ─── Section 1: Property Details ──────────────────────────────── */}
   436→        <section>
   437→          <h2 className={sectionHeaderClass}>Property Details</h2>
   438→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">

   457→                name="owner"
   458→                value={form.owner}
   459→                onChange={handleFormChange}

   461→                className={inputClass}

   465→            {/* Owner Contact */}
