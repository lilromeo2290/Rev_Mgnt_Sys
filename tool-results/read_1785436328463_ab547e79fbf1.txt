   430→                      </div>
   431→                    </td>
   432→                    <td className="px-4 py-3">
   433→                      <div>
   434→                        <p className="text-sm font-medium text-slate-900 dark:text-white">{r.issuedTo}</p>
   435→                        <p className="text-xs text-slate-500 dark:text-slate-400">{r.entityType}</p>
   436→                      </div>
   437→                    </td>
   438→                    <td className="px-4 py-3 text-sm text-slate-600 dark:text-slate-400">{r.revenueItem}</td>
   439→                    <td className="px-4 py-3 text-sm font-medium text-slate-900 dark:text-white text-right">{fmtCurrency(r.totalPaid)}</td>
   440→                    <td className="px-4 py-3">
   441→                      <span className={`inline-flex text-xs font-medium px-2 py-0.5 rounded-full ${methodStyle[r.method]}`}>
   442→                        {r.method}
   443→                      </span>
   444→                    </td>
   445→                    <td className="px-4 py-3">
   446→                      <div className="text-sm text-slate-600 dark:text-slate-400">{r.date}</div>
   447→                      <div className="text-xs text-slate-400 dark:text-slate-500">{r.time}</div>
   448→                    </td>
   449→                    <td className="px-4 py-3">
   450→                      <span className={`inline-flex text-xs font-medium px-2 py-0.5 rounded-full ${statusStyle[r.status].pill}`}>
   451→                        {r.status}
   452→                      </span>
   453→                    </td>
   454→                    <td className="px-4 py-3">
   455→                      <div className="flex items-center justify-center gap-1">
   456→                        <button
   457→                          onClick={() => setSelectedReceipt(r)}
   458→                          className="p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   459→                          title="View details"
   460→                        >
   461→                          <Eye className="w-4 h-4" />
   462→                        </button>
   463→                        <button
   464→                          onClick={() => handlePrintReceipt(r)}
   465→                          className="p-1.5 rounded-lg text-slate-400 hover:text-emerald-600 dark:hover:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 transition-colors"
   466→                          title="Print receipt"
   467→                        >
   468→                          <Printer className="w-4 h-4" />
   469→                        </button>
   470→                      </div>
   471→                    </td>
   472→                  </tr>
   473→                ))
   474→              )}
   475→            </tbody>
   476→          </table>
   477→        </div>
   478→
   479→        {/* Pagination */}
   480→        <div className="flex items-center justify-between px-4 py-3 border-t border-slate-200 dark:border-slate-700">
   481→          <p className="text-sm text-slate-500 dark:text-slate-400">
   482→            Showing {paginated.length} of {filtered.length} receipts
   483→          </p>
   484→          <div className="flex items-center gap-1">
   485→            <button
   486→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   487→              disabled={page === 1}
   488→              className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   489→            >
   490→              <ChevronLeft className="w-4 h-4" />
   491→            </button>
   492→            {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
   493→              <button
   494→                key={p}
   495→                onClick={() => setPage(p)}
   496→                className={`w-8 h-8 rounded-lg text-sm font-medium transition-colors ${
   497→                  p === page
   498→                    ? 'bg-emerald-600 text-white'
   499→                    : 'text-slate-500 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-700'
   500→                }`}
   501→              >
   502→                {p}
   503→              </button>
   504→            ))}
   505→            <button
   506→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   507→              disabled={page === totalPages}
   508→              className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 disabled:opacity-40 disabled:cursor-not-allowed transition-colors"
   509→            >
   510→              <ChevronRight className="w-4 h-4" />
   511→            </button>
   512→          </div>
   513→        </div>
   514→      </div>
   515→
   516→      {/* Detail Modal */}
   517→      {selectedReceipt && (
   518→        <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
   519→          <div className="fixed inset-0 bg-black/50 backdrop-blur-sm" onClick={() => setSelectedReceipt(null)} />
   520→          <div className="relative bg-white dark:bg-slate-800 rounded-xl border border-slate-200 dark:border-slate-700 shadow-2xl w-full max-w-lg max-h-[90vh] overflow-y-auto">
   521→            {/* Modal Header */}
   522→            <div className="flex items-center justify-between p-5 border-b border-slate-200 dark:border-slate-700">
   523→              <div className="flex items-center gap-3">
   524→                <div className="w-10 h-10 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   525→                  <Receipt className="w-5 h-5 text-emerald-600 dark:text-emerald-400" />
   526→                </div>
   527→                <div>
   528→                  <h2 className="text-lg font-semibold text-slate-900 dark:text-white">Receipt Details</h2>
   529→                  <p className="text-sm font-mono text-slate-500 dark:text-slate-400">{selectedReceipt.receiptNo}</p>
   530→                </div>
   531→              </div>
   532→              <button
   533→                onClick={() => setSelectedReceipt(null)}
   534→                className="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300 hover:bg-slate-100 dark:hover:bg-slate-700 transition-colors"
   535→              >
   536→                <X className="w-5 h-5" />
   537→              </button>
   538→            </div>
   539→
   540→            {/* Modal Body */}
   541→            <div className="p-5 space-y-5">
   542→              {/* Status Badge */}
   543→              <div className="flex items-center justify-between">
   544→                <span className={`inline-flex items-center gap-1.5 text-xs font-semibold px-2.5 py-1 rounded-full ${statusStyle[selectedReceipt.status].pill}`}>
   545→                  {selectedReceipt.status}
   546→                </span>
   547→                <div className="flex items-center gap-2">
   548→                  <button onClick={() => handlePrintReceipt(selectedReceipt)} className="inline-flex items-center gap-1.5 rounded-lg bg-emerald-600 text-white px-3 py-1.5 text-sm font-medium hover:bg-emerald-700 transition-colors cursor-pointer">
   549→                    <Printer className="w-3.5 h-3.5" />
   550→                    Print
   551→                  </button>
   552→                  <button onClick={() => handlePrintReceipt(selectedReceipt)} className="inline-flex items-center gap-1.5 rounded-lg border border-slate-200 dark:border-slate-700 text-slate-700 dark:text-slate-300 px-3 py-1.5 text-sm font-medium hover:bg-slate-50 dark:hover:bg-slate-700 transition-colors cursor-pointer">
   553→                    <Download className="w-3.5 h-3.5" />
   554→                    PDF
   555→                  </button>
   556→                </div>
   557→              </div>
   558→
   559→              {/* Info Grid */}
   560→              <div className="grid grid-cols-2 gap-4">
   561→                <InfoField label="Bill Reference" value={selectedReceipt.billNo} />
   562→                <InfoField label="Payment Method" value={selectedReceipt.method} />
   563→                <InfoField label="Transaction Ref" value={selectedReceipt.reference} />
   564→                <InfoField label="Zone / Ward" value={`${selectedReceipt.zone} — ${selectedReceipt.ward}`} />
   565→                <InfoField label="Collected By" value={selectedReceipt.issuedBy} />
   566→                <InfoField label="Entity Type" value={selectedReceipt.entityType} />
   567→              </div>
   568→
   569→              {/* Issued To */}
   570→              <div className="rounded-lg bg-slate-50 dark:bg-slate-900/50 p-4">
   571→                <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium mb-1">Issued To</p>
   572→                <p className="text-lg font-semibold text-slate-900 dark:text-white">{selectedReceipt.issuedTo}</p>
   573→              </div>
   574→
   575→              {/* Line Items */}
   576→              <div>
   577→                <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium mb-3">Line Items</p>
   578→                <div className="rounded-lg border border-slate-200 dark:border-slate-700 overflow-hidden">
   579→                  <table className="w-full text-left text-sm">
   580→                    <thead className="bg-slate-50 dark:bg-slate-900/50">
   581→                      <tr className="border-b border-slate-200 dark:border-slate-700">
   582→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400">Description</th>
   583→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Qty</th>
   584→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Unit Price</th>
   585→                        <th className="px-3 py-2 text-xs font-medium text-slate-500 dark:text-slate-400 text-right">Amount</th>
   586→                      </tr>
   587→                    </thead>
   588→                    <tbody className="divide-y divide-slate-100 dark:divide-slate-700/60">
   589→                      {selectedReceipt.items.map((item, idx) => (
   590→                        <tr key={idx}>
   591→                          <td className="px-3 py-2 text-slate-700 dark:text-slate-300">{item.description}</td>

   595→                        </tr>
   596→                      ))}
   597→                    </tbody>
   598→                    <tfoot className="bg-slate-50 dark:bg-slate-900/50">
   599→                      <tr className="border-t border-slate-200 dark:border-slate-700">
   600→                        <td colSpan={3} className="px-3 py-2.5 text-sm font-semibold text-slate-700 dark:text-slate-300 text-right">Total Paid</td>
   601→                        <td className="px-3 py-2.5 text-sm font-bold text-emerald-600 dark:text-emerald-400 text-right">{fmtCurrency(selectedReceipt.totalPaid)}</td>
   602→                      </tr>
   603→                    </tfoot>
   604→                  </table>
   605→                </div>
   606→              </div>
   607→
   608→              {/* Barcode Section */}
   609→              <div className="rounded-lg border border-slate-200 dark:border-slate-700 p-4">
   610→                <div className="flex items-center gap-2 mb-3">
   611→                  <ScanBarcode className="w-4 h-4 text-slate-500 dark:text-slate-400" />
   612→                  <p className="text-xs uppercase text-slate-500 dark:text-slate-400 font-medium">Verification Barcode</p>
   613→                </div>
   614→                <div className="flex flex-col items-center bg-white dark:bg-slate-900 rounded-lg p-3">
   615→                  <canvas ref={barcodeCanvasRef} className="max-w-full" />
   616→                  <p className="text-[9px] text-slate-400 dark:text-slate-500 mt-2 text-center break-all">
   617→                    Scan barcode or visit verify page to confirm authenticity

   619→                </div>
   620→              </div>
   621→
   622→              {/* Date / Time */}
   623→              <div className="flex items-center gap-6 text-sm text-slate-500 dark:text-slate-400">

   626→              </div>
   627→            </div>
   628→          </div>
   629→        </div>
   630→      )}
   631→    </div>
   632→  );
   633→}
